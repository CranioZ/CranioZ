"""CranioZ application entry point.

Boot sequence:
1. Load configuration.
2. Configure logging.
3. Initialize the project path resolver.
4. Initialize the event bus.
5. Start the Qt application loop.

The main window is not created in this phase.
"""

from __future__ import annotations

import logging
import logging.config
from pathlib import Path

import yaml
from PySide6.QtWidgets import QApplication

from cranioz import __version__
from cranioz.application.event_bus import EventBus
from cranioz.config.loader import ConfigurationLoader
from cranioz.config.schema import Config
from cranioz.infrastructure.app.app_path_resolver import AppPathResolver

_logger = logging.getLogger(__name__)

_DEFAULT_LOGGING_CONFIG = Path(__file__).parent / "config" / "defaults" / "logging.yaml"


def _configure_logging(config: Config) -> None:
    """Configure logging from the packaged YAML, then apply config overrides."""
    if _DEFAULT_LOGGING_CONFIG.exists():
        with _DEFAULT_LOGGING_CONFIG.open("r", encoding="utf-8") as fh:
            logging.config.dictConfig(yaml.safe_load(fh))
    logging.getLogger().setLevel(config.logging.level)


def main() -> int:
    """Run the CranioZ application."""
    config = ConfigurationLoader().load()
    _configure_logging(config)
    _logger.info("CranioZ %s starting", __version__)

    app_paths = AppPathResolver()
    app_paths.initialize(config.paths)
    _logger.info("Projects root: %s", app_paths.projects_root)

    event_bus = EventBus()
    _logger.debug("Event bus initialized: %r", event_bus)

    app = QApplication.instance() or QApplication([])
    _logger.info("Boot complete")
    return app.exec()


if __name__ == "__main__":
    raise SystemExit(main())

"""Application event bus.

Minimal synchronous publish/subscribe bus for application-level events.
Threading and async dispatch are out of scope for this phase.

Rules:
- Subscribers receive events synchronously, in registration order.
- Publishing an event with no subscribers is a no-op.
- A failing subscriber does not prevent other subscribers from running;
  errors are collected and re-raised after all subscribers have been
  notified.
"""

from __future__ import annotations

from collections import defaultdict
from collections.abc import Callable
from typing import Any


class EventBusError(Exception):
    """Raised when subscriber dispatch fails."""


EventHandler = Callable[[Any], None]


class EventBus:
    """Synchronous in-process event bus."""

    def __init__(self) -> None:
        self._subscribers: dict[type, list[EventHandler]] = defaultdict(list)

    def subscribe(self, event_type: type, handler: EventHandler) -> None:
        """Register a handler for a specific event type."""
        self._subscribers[event_type].append(handler)

    def unsubscribe(self, event_type: type, handler: EventHandler) -> None:
        """Remove a previously registered handler.

        No-op if the handler is not registered.
        """
        handlers = self._subscribers.get(event_type)
        if not handlers:
            return
        try:
            handlers.remove(handler)
        except ValueError:
            return

    def publish(self, event: Any) -> None:
        """Publish an event to all handlers registered for its type.

        Dispatches in registration order. If any handler raises, the
        remaining handlers still run, and the collected exceptions are
        re-raised as an EventBusError at the end.
        """
        handlers = list(self._subscribers.get(type(event), ()))
        errors: list[BaseException] = []
        for handler in handlers:
            try:
                handler(event)
            except Exception as exc:
                errors.append(exc)
        if errors:
            raise EventBusError(
                f"{len(errors)} subscriber(s) failed while publishing {type(event).__name__}"
            ) from errors[0]

    def subscriber_count(self, event_type: type) -> int:
        """Return the number of handlers registered for an event type."""
        return len(self._subscribers.get(event_type, ()))

"""ConfigurationLoader.

Loads system configuration from src/cranioz/config/defaults/.
Does not load user preferences; that is the responsibility of settings/.

Rules:
- The loader reads the defaults packaged with the application.
- The loader does not resolve physical OS paths.
- The loader returns a validated Config.
"""

from __future__ import annotations

import json
from pathlib import Path
from typing import Any

from cranioz.config.schema import Config, ConfigError, build_config

_DEFAULTS_DIR = Path(__file__).parent / "defaults"


class ConfigurationLoader:
    """Loads and validates the system configuration."""

    def __init__(self, defaults_dir: Path | None = None) -> None:
        self._defaults_dir = defaults_dir or _DEFAULTS_DIR

    def load(self) -> Config:
        """Load app.json and paths.json, validate, and return a Config."""
        app_data = self._read_json("app.json")
        paths_data = self._read_json("paths.json")

        try:
            return build_config(app_data, paths_data)
        except ConfigError:
            raise
        except Exception as exc:
            raise ConfigError(f"failed to build configuration: {exc}") from exc

    def _read_json(self, filename: str) -> dict[str, Any]:
        path = self._defaults_dir / filename
        if not path.exists():
            raise ConfigError(f"configuration file not found: {path}")

        try:
            with path.open("r", encoding="utf-8") as fh:
                data = json.load(fh)
        except json.JSONDecodeError as exc:
            raise ConfigError(f"invalid JSON in {path}: {exc}") from exc

        if not isinstance(data, dict):
            raise ConfigError(f"unexpected content in {path}: expected object")
        return data

"""Configuration schema migration.

In this phase, the schema is 1 and there are no migrations.
This class exists as a future extension point.
"""

from __future__ import annotations

from typing import Any

CURRENT_SCHEMA_VERSION = 1


class ConfigMigrationError(Exception):
    """Error during configuration migration."""


class ConfigMigrator:
    def migrate(self, data: dict[str, Any], from_version: int) -> dict[str, Any]:
        if from_version == CURRENT_SCHEMA_VERSION:
            return data

        if from_version > CURRENT_SCHEMA_VERSION:
            raise ConfigMigrationError(
                f"schema {from_version} is newer than supported ({CURRENT_SCHEMA_VERSION})"
            )

        # Extension point for future migrations.
        raise ConfigMigrationError(
            f"no migration defined from schema {from_version} to {CURRENT_SCHEMA_VERSION}"
        )

"""Configuration schema and validation.

Defines required keys and accepted types.
No external dependencies; explicit validation only.
"""

from __future__ import annotations

from dataclasses import dataclass
from typing import Any


class ConfigError(Exception):
    """Configuration error: missing key, invalid type, or inconsistent value."""


@dataclass(frozen=True)
class AppConfig:
    name: str
    language: str
    theme: str
    recent_projects_limit: int


@dataclass(frozen=True)
class PathsConfig:
    project_root: str | None
    flows_root: str | None
    cache_root: str | None


@dataclass(frozen=True)
class LoggingConfig:
    level: str
    format: str


@dataclass(frozen=True)
class Config:
    app: AppConfig
    paths: PathsConfig
    logging: LoggingConfig


def _require(data: dict[str, Any], key: str, context: str) -> Any:
    if key not in data:
        raise ConfigError(f"missing key: {context}.{key}")
    return data[key]


def _require_type(value: Any, expected: type | tuple[type, ...], context: str) -> None:
    if not isinstance(value, expected):
        raise ConfigError(f"invalid type at {context}: expected {expected}, got {type(value)}")


def build_config(app_data: dict[str, Any], paths_data: dict[str, Any]) -> Config:
    app_section = _require(app_data, "app", "root")
    _require_type(app_section, dict, "root.app")

    logging_section = _require(app_data, "logging", "root")
    _require_type(logging_section, dict, "root.logging")

    paths_section = _require(paths_data, "paths", "root")
    _require_type(paths_section, dict, "root.paths")

    app = AppConfig(
        name=_require(app_section, "name", "app"),
        language=_require(app_section, "language", "app"),
        theme=_require(app_section, "theme", "app"),
        recent_projects_limit=_require(app_section, "recent_projects_limit", "app"),
    )

    _require_type(app.recent_projects_limit, int, "app.recent_projects_limit")

    logging_cfg = LoggingConfig(
        level=_require(logging_section, "level", "logging"),
        format=_require(logging_section, "format", "logging"),
    )

    paths = PathsConfig(
        project_root=paths_section.get("project_root"),
        flows_root=paths_section.get("flows_root"),
        cache_root=paths_section.get("cache_root"),
    )

    return Config(app=app, paths=paths, logging=logging_cfg)

"""Patient.

Aggregate root of the Patient domain.

A Patient is the person associated with clinical data in CranioZ.
Within the CranioZ architecture, there is no separate PatientManager:
the patient's lifecycle is managed through the Project Manager, in
the context of the projects that reference the patient.

The Patient is composed of nine section value objects, one per
functional section of the patient record (see ADR-0004):

    Patient
    ├── identity:        PatientIdentity          (§1 Identification)
    ├── documents:       PatientDocuments         (§2 Documents and Contact)
    ├── filiation:       PatientFiliation         (§3 Parentage)
    ├── address:         PatientAddress           (§4 Address)
    ├── complaint:       PatientComplaint         (§5 Complaint and History)
    ├── comorbidities:   PatientComorbidities     (§6 Comorbidities)
    ├── medications:     PatientMedications       (§7 Medications and Allergies)
    ├── antecedents:     PatientAntecedents       (§8 Medical History)
    └── habits:          PatientHabits            (§9 Habits and Addictions)

Design rules (see ADR-0004 §2 and §5):

- The Patient carries the PatientId, the nine section value objects,
  and the cross-section invariants.
- Cross-section invariants are enforced by this aggregate, not by
  the sections. Sections are responsible only for the invariants
  they can verify in isolation.
- The Patient does not carry references to imaging, meshes,
  photographs, odontogram, or documents. Those entities reference
  the Patient, not the other way around.
- There is no ``extras`` dict. If a future piece genuinely needs a
  free-form escape hatch, it will be reintroduced with a specific
  purpose.
- The Patient is mutable, because it is the entity through which
  updates are coordinated. The section value objects are immutable.

Cross-section invariants enforced here:

1. Women's health applicability. The fields
   ``antecedents.pregnancy_contraception`` and
   ``antecedents.breastfeeding`` are meaningful only when
   ``identity.sex is PatientSex.FEMALE``. When the sex is not
   FEMALE (including ``None``), both fields must be empty.

Document coherence and insurance coherence are enforced by
``PatientDocuments`` itself, because they depend only on fields
inside that section.
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import date

from cranioz.domain.patient.enums.patient_sex import PatientSex
from cranioz.domain.patient.patient_summary import PatientSummary
from cranioz.domain.patient.sections.address import PatientAddress
from cranioz.domain.patient.sections.antecedents import PatientAntecedents
from cranioz.domain.patient.sections.comorbidities import (
    PatientComorbidities,
)
from cranioz.domain.patient.sections.complaint import PatientComplaint
from cranioz.domain.patient.sections.documents import PatientDocuments
from cranioz.domain.patient.sections.filiation import PatientFiliation
from cranioz.domain.patient.sections.habits import PatientHabits
from cranioz.domain.patient.sections.identity import PatientIdentity
from cranioz.domain.patient.sections.medications import PatientMedications
from cranioz.domain.patient.value_objects.patient_id import PatientId


@dataclass
class Patient:
    """Aggregate root of the Patient domain."""

    patient_id: PatientId
    identity: PatientIdentity
    documents: PatientDocuments
    filiation: PatientFiliation
    address: PatientAddress
    complaint: PatientComplaint
    comorbidities: PatientComorbidities
    medications: PatientMedications
    antecedents: PatientAntecedents
    habits: PatientHabits

    # ----- Validation -------------------------------------------------

    def __post_init__(self) -> None:
        self._validate_womens_health_applicability()

    def _validate_womens_health_applicability(self) -> None:
        """Enforce the cross-section invariant on women's health.

        The fields of ``antecedents.pregnancy_contraception`` and
        ``antecedents.breastfeeding`` are meaningful only when the
        patient's sex is ``FEMALE``. When the sex is not FEMALE
        (including ``None``), both fields must be empty.
        """
        sex = self.identity.sex
        if sex is PatientSex.FEMALE:
            return

        if self.antecedents.pregnancy_contraception:
            raise ValueError(
                "antecedents.pregnancy_contraception must be empty when identity.sex is not FEMALE"
            )
        if self.antecedents.breastfeeding:
            raise ValueError(
                "antecedents.breastfeeding must be empty when identity.sex is not FEMALE"
            )

    # ----- Convenience -----------------------------------------------

    def display_name(self) -> str:
        """Return the name used to present the patient in the UI.

        The rule (ADR-0004 §13): prefer ``identity.social_name``
        when set, and fall back to ``identity.full_name``.
        """
        if self.identity.social_name:
            return self.identity.social_name
        return self.identity.full_name

    def age(self, at_date: date | None = None) -> int | None:
        """Return the patient's age in years.

        Returns ``None`` when the birth date is not recorded.

        The reference date defaults to today. The computation
        accounts for whether the birthday has already occurred in
        the reference year: a person born on 2000-12-31 is 25 on
        2026-01-01, not 26.
        """
        if self.identity.birth_date is None:
            return None
        ref = at_date if at_date is not None else date.today()
        years = ref.year - self.identity.birth_date.year
        # Subtract one if the birthday has not yet occurred in ref.
        if (ref.month, ref.day) < (
            self.identity.birth_date.month,
            self.identity.birth_date.day,
        ):
            years -= 1
        return years

    def summary(self) -> PatientSummary:
        """Return a lightweight projection of this Patient.

        See ``PatientSummary`` for the derivation rules.
        """
        return PatientSummary(
            patient_id=self.patient_id,
            display_name=self.display_name(),
            birth_date=self.identity.birth_date,
            sex=self.identity.sex,
        )

    # ----- Representation --------------------------------------------

    def __repr__(self) -> str:
        return (
            f"Patient(patient_id={self.patient_id.value!r}, display_name={self.display_name()!r})"
        )

    def __eq__(self, other: object) -> bool:
        """Two patients are equal if they share the same PatientId.

        This makes Patient usable in sets and as dict keys, and it
        is consistent with the identity semantics of a domain
        entity. The value objects inside the aggregate are
        compared by value (frozen dataclasses), so the id is the
        only discriminating field that matters at the aggregate
        level.
        """
        if not isinstance(other, Patient):
            return NotImplemented
        return self.patient_id == other.patient_id

    def __hash__(self) -> int:
        """Hash based on PatientId, consistent with ``__eq__``."""
        return hash(self.patient_id)

"""PatientSummary.

Lightweight read-only projection of a Patient.

The PatientSummary is designed for contexts where the full Patient
record is not needed: project listings, tooltips, breadcrumbs, and
the ProjectContext handed to the Workspace. It carries only the
fields required to *identify* and *display* the patient without
loading their complete clinical record.

The summary is a value object. It is derived from a Patient but is
not the Patient itself, and it does not carry any clinical data
(no medical history, no imaging references).

Derivation rules (see ADR-0004 §13):

- ``display_name`` is produced by ``Patient.display_name()``, which
  prefers ``identity.social_name`` when set, and falls back to
  ``identity.full_name``.
- ``birth_date`` and ``sex`` are taken directly from
  ``Patient.identity``.

The summary is a frozen value object. It is not updated when the
Patient changes; callers that need the current state must call
``Patient.summary()`` again.
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import date

from cranioz.domain.patient.enums.patient_sex import PatientSex
from cranioz.domain.patient.value_objects.patient_id import PatientId


@dataclass(frozen=True)
class PatientSummary:
    """Lightweight read-only projection of a Patient."""

    patient_id: PatientId
    display_name: str
    birth_date: date | None
    sex: PatientSex | None

    def __post_init__(self) -> None:
        if not self.display_name:
            raise ValueError("display_name must not be empty")
        if not self.display_name.strip():
            raise ValueError("display_name must not be whitespace only")

"""BloodType.

Blood type of the patient, combining ABO group and Rh factor.

The domain uses a single value for the combination because the two
are always recorded together, always used together, and rarely
meaningful apart. Storing them as two fields would invite
inconsistent states, such as an ABO group without an Rh factor, or
an Rh factor without an ABO group.

``UNKNOWN`` is a valid value, distinct from "not applicable". At
the first consultation, the blood type may not yet be available.
The field is required at the domain level and defaults to
``UNKNOWN`` so that "not recorded" is an explicit value in the
persisted record, not a missing field.

The enum identifiers and values are in English, consistent with
the rest of the CranioZ codebase. The user-facing labels
("A+", "A-", "O+", "Desconhecido", ...) are provided by the i18n
layer, keyed by the enum value.

See ADR-0004 §4 for the design rationale.
"""

from __future__ import annotations

from enum import Enum


class BloodType(Enum):
    """Blood type combining ABO group and Rh factor."""

    A_POSITIVE = "a_positive"
    A_NEGATIVE = "a_negative"
    B_POSITIVE = "b_positive"
    B_NEGATIVE = "b_negative"
    AB_POSITIVE = "ab_positive"
    AB_NEGATIVE = "ab_negative"
    O_POSITIVE = "o_positive"
    O_NEGATIVE = "o_negative"
    UNKNOWN = "unknown"

"""DocumentType.

Kind of identity document recorded for the patient, expressed as
a generic concept that is independent of the issuing country.

The domain stores the type, the country, and the number together
(in ``PatientDocuments``). It does not validate the format of the
number: that is a localization concern that belongs to the UI
layer (mask, validation messages).

The document type is deliberately generic:

- ``NATIONAL_ID`` covers RG (BR), DNI (ES), and equivalent
  national identity documents.
- ``TAX_ID`` covers CPF (BR), SSN (US), NIF (PT), and equivalent
  tax identification numbers.
- ``DRIVERS_LICENSE`` covers CNH (BR) and equivalent driving
  licenses.

The country-specific name and format (RG vs. DNI, CPF vs. SSN) is
resolved by the UI layer from the ``(document_type, country)``
pair, using the i18n catalogs. The domain never stores the
localized name as a technical identifier.

This design keeps the persisted representation stable across
locales and countries: a Brazilian patient's ``tax_id`` remains
``tax_id`` regardless of the active language or deployment country.

See ADR-0004 §5 for the design rationale.
"""

from __future__ import annotations

from enum import Enum


class DocumentType(Enum):
    """Generic kind of identity document, without country-specific encoding."""

    NATIONAL_ID = "national_id"
    TAX_ID = "tax_id"
    DRIVERS_LICENSE = "drivers_license"
    PASSPORT = "passport"
    BIRTH_CERTIFICATE = "birth_certificate"
    OTHER = "other"

FILE: src\cranioz\domain\patient\sections\documents.py

"""PatientDocuments (§2 Documentos e Contato).

Identity documents, contact information, and health insurance data.

The section carries:

- One identity document (type, country, number).
- Two phone numbers (primary and optional secondary), each with a
  kind.
- An optional email.
- Optional health insurance data (name and card number).

Conditional rules enforced by this section:

- If ``document_type`` is set, both ``document_country`` and
  ``document_number`` must be set and non-empty.
- If ``document_type`` is ``None``, both ``document_country`` and
  ``document_number`` must be ``None``.
- ``document_country``, when set, must be an ISO 3166-1 alpha-2
  code: two uppercase ASCII letters (e.g., ``"BR"``, ``"US"``,
  ``"PT"``).
- If ``health_insurance`` is set, ``insurance_card_number`` must be
  set and non-empty.
- If ``health_insurance`` is ``None``, ``insurance_card_number``
  must be ``None``.

These rules depend only on fields inside this section, so they are
enforced here, not in the Patient aggregate.

See ADR-0004 §5 for the design rationale.
"""

from __future__ import annotations

from dataclasses import dataclass

from cranioz.domain.patient.enums.document_type import DocumentType
from cranioz.domain.patient.value_objects.phone_number import PhoneNumber


@dataclass(frozen=True)
class PatientDocuments:
    """Documents and contact information of the patient (§2)."""

    document_type: DocumentType | None = None
    document_country: str | None = None
    document_number: str | None = None
    phone_primary: PhoneNumber | None = None
    phone_secondary: PhoneNumber | None = None
    email: str | None = None
    health_insurance: str | None = None
    insurance_card_number: str | None = None
    notes: str = ""

    def __post_init__(self) -> None:
        self._validate_document()
        self._validate_insurance()

    def _validate_document(self) -> None:
        if self.document_type is None:
            if self.document_country is not None:
                raise ValueError("document_country must be None when document_type is None")
            if self.document_number is not None:
                raise ValueError("document_number must be None when document_type is None")
            return

        if not self.document_country:
            raise ValueError("document_country is required when document_type is set")
        if not self.document_number:
            raise ValueError("document_number is required when document_type is set")
        if not self.document_number.strip():
            raise ValueError("document_number must not be whitespace only")

        # ISO 3166-1 alpha-2: exactly two uppercase ASCII letters.
        if (
            len(self.document_country) != 2
            or not self.document_country.isascii()
            or not self.document_country.isupper()
        ):
            raise ValueError(
                "document_country must be an ISO 3166-1 alpha-2 code "
                f"(two uppercase ASCII letters), got {self.document_country!r}"
            )

    def _validate_insurance(self) -> None:
        if self.health_insurance is None:
            if self.insurance_card_number is not None:
                raise ValueError("insurance_card_number must be None when health_insurance is None")
            return

        if not self.health_insurance.strip():
            raise ValueError("health_insurance must not be whitespace only")
        if not self.insurance_card_number:
            raise ValueError("insurance_card_number is required when health_insurance is set")
        if not self.insurance_card_number.strip():
            raise ValueError("insurance_card_number must not be whitespace only")

"""Project.

Aggregate root of the Project domain.

A Project is the persistent clinical container of CranioZ. It
references a Patient and holds the metadata, clinical data
references, workflow state, and configuration of one planning case.

This class is deliberately minimal in this piece. It carries the
project's identity and metadata, and it defines the invariant that
the metadata's project_id matches the Project's own project_id. As
more domain concepts are implemented (patient record, workflow
state, object references), they will be added to this aggregate.

The Project is mutable because it is the aggregate root through
which the Project Manager coordinates updates. Individual value
objects (ProjectId, ProjectMetadata) are immutable; the aggregate
is the boundary across which consistency is maintained.

Rules:

- The Project always has a ProjectId and a ProjectMetadata.
- The metadata's project_id must equal the Project's project_id.
- The Project does not know about persistence, filesystem, or Qt.
  Those are responsibilities of the infrastructure and application
  layers.
"""

from __future__ import annotations

from dataclasses import dataclass, field

from cranioz.domain.project.project_id import ProjectId
from cranioz.domain.project.project_metadata import ProjectMetadata


@dataclass
class Project:
    """Aggregate root of the Project domain."""

    project_id: ProjectId
    metadata: ProjectMetadata
    # Extension point: further fields (patient_record, workflow_state,
    # object_references, ...) will be added as they are implemented.
    # Kept as a plain dict for forward compatibility of serialization.
    extras: dict[str, object] = field(default_factory=dict)

    def __post_init__(self) -> None:
        if self.metadata.project_id != self.project_id:
            raise ValueError(
                "metadata.project_id must match project_id "
                f"(metadata={self.metadata.project_id.value!r}, "
                f"project={self.project_id.value!r})"
            )

    def update_metadata(self, new_metadata: ProjectMetadata) -> None:
        """Replace the project's metadata.

        Raises ValueError if the new metadata belongs to a different
        project. This protects the aggregate invariant at every
        update boundary.
        """
        if new_metadata.project_id != self.project_id:
            raise ValueError(
                "cannot replace metadata with one belonging to a "
                f"different project (expected {self.project_id.value!r}, "
                f"got {new_metadata.project_id.value!r})"
            )
        self.metadata = new_metadata

    def __repr__(self) -> str:
        return (
            f"Project(project_id={self.project_id.value!r}, "
            f"name={self.metadata.project_name!r}, "
            f"status={self.metadata.status.value!r})"
        )

FILE: src\cranioz\domain\project\project_metadata.py
======================================================================

"""ProjectMetadata.

Persistent metadata of a Project. This is the lightweight
information that describes *what* the project is without loading
its clinical data.

The metadata is used by:

- The Project Catalog, to render project lists without loading the
  full project.
- The Project Manager, to reason about the project's status,
  schema version, and lifecycle.
- The Project Repository, as the top-level structure that describes
  the persistent representation.

The metadata is immutable. Updates produce a new instance; the
Project Manager is responsible for coordinating those updates and
for persisting them through the repository.

Timestamps use timezone-aware datetime objects in UTC. The domain
does not concern itself with serialization; the infrastructure
layer is responsible for converting to and from ISO 8601.
"""

from __future__ import annotations

from dataclasses import dataclass, replace
from datetime import datetime

from cranioz.domain.project.project_id import ProjectId
from cranioz.domain.project.project_status import ProjectStatus


@dataclass(frozen=True)
class ProjectMetadata:
    """Persistent metadata of a Project."""

    project_id: ProjectId
    project_name: str
    patient_id: str
    patient_display_name: str
    clinical_procedure: str
    status: ProjectStatus
    created_at: datetime
    updated_at: datetime
    last_opened_at: datetime | None
    schema_version: int

    def __post_init__(self) -> None:
        if not self.project_name:
            raise ValueError("project_name must not be empty")
        if not self.patient_id:
            raise ValueError("patient_id must not be empty")
        if not self.patient_display_name:
            raise ValueError("patient_display_name must not be empty")
        if self.schema_version < 1:
            raise ValueError(f"schema_version must be >= 1, got {self.schema_version}")

        # Validate timezone-awareness BEFORE comparing datetimes.
        # Comparing a naive and an aware datetime raises TypeError
        # in Python; we want a clear ValueError instead.
        self._validate_timezone(self.created_at, "created_at")
        self._validate_timezone(self.updated_at, "updated_at")
        if self.last_opened_at is not None:
            self._validate_timezone(self.last_opened_at, "last_opened_at")

        if self.updated_at < self.created_at:
            raise ValueError("updated_at must not be earlier than created_at")
        if self.last_opened_at is not None and self.last_opened_at < self.created_at:
            raise ValueError("last_opened_at must not be earlier than created_at")

    @staticmethod
    def _validate_timezone(value: datetime, field_name: str) -> None:
        if value.tzinfo is None:
            raise ValueError(f"{field_name} must be timezone-aware (UTC)")

    # ----- Convenience mutators -------------------------------------

    def with_status(
        self,
        status: ProjectStatus,
        *,
        now: datetime,
    ) -> ProjectMetadata:
        ...
        """Return a new metadata with the given status.

        ``now`` becomes the new ``updated_at``. The caller is
        responsible for supplying a timezone-aware UTC datetime.
        """
        return replace(self, status=status, updated_at=now)

    def with_name(
        self,
        project_name: str,
        *,
        now: datetime,
    ) -> ProjectMetadata:
        """Return a new metadata with a new project_name."""
        if not project_name:
            raise ValueError("project_name must not be empty")
        return replace(self, project_name=project_name, updated_at=now)

    def with_last_opened(
        self,
        *,
        now: datetime,
    ) -> ProjectMetadata:
        """Return a new metadata with last_opened_at set to now."""
        return replace(self, last_opened_at=now)

"""AppPathResolver.

Resolves the application-level paths of CranioZ.

The resolver answers questions like:

- Where is the root directory that contains all projects?
- Where are the flow definitions stored?
- Where is the cache stored?

These paths are configured once at boot from ``paths.json`` and
remain stable for the entire session.

The resolver does NOT know about individual projects. It only
provides the roots that other components use to locate data. The
paths of a specific project are resolved by the ProjectPathResolver
(see ``infrastructure/project/project_path_resolver.py``).

Rules:

- ``initialize`` accepts a ``PathsConfig`` and resolves each root.
- If a path is configured, it is used.
- If a path is ``None``, a default is derived from the user's home
  directory.
- Missing directories are created unless ``create=False`` is passed.
- Failure to resolve or create a directory raises
  ``AppPathInaccessible``.
"""

from __future__ import annotations

from pathlib import Path

from cranioz.config.schema import PathsConfig


class AppPathInaccessible(Exception):
    """Raised when an application path cannot be resolved, created, or accessed."""


DEFAULT_PROJECT_ROOT_NAME = "CranioZ"
DEFAULT_PROJECTS_DIR_NAME = "Projects"
DEFAULT_FLOWS_DIR_NAME = "Flows"
DEFAULT_CACHE_DIR_NAME = "Cache"


class AppPathResolver:
    """Resolves the application-level paths of CranioZ.

    Configured once at boot. Provides the roots that other components
    use to locate data. Does not know about ProjectId or project
    structure.
    """

    def __init__(self) -> None:
        self._projects_root: Path | None = None
        self._flows_root: Path | None = None
        self._cache_root: Path | None = None

    # ----- Initialization -------------------------------------------

    def initialize(
        self,
        config: PathsConfig,
        *,
        create: bool = True,
    ) -> None:
        """Resolve and store the application paths from configuration.

        Args:
            config: the ``PathsConfig`` section of the application
                configuration. May contain ``None`` for any path, in
                which case the default is used.
            create: if True, create missing directories. If False,
                missing directories raise ``AppPathInaccessible``.

        Raises:
            AppPathInaccessible: if any path cannot be resolved,
                created, or accessed.
        """
        self._projects_root = self._resolve_dir(
            configured=config.project_root,
            default=self._default_projects_root(),
            field_name="project_root",
            create=create,
        )
        self._flows_root = self._resolve_dir(
            configured=config.flows_root,
            default=self._default_flows_root(),
            field_name="flows_root",
            create=create,
        )
        self._cache_root = self._resolve_dir(
            configured=config.cache_root,
            default=self._default_cache_root(),
            field_name="cache_root",
            create=create,
        )

    # ----- Accessors ------------------------------------------------

    @property
    def projects_root(self) -> Path:
        """Directory that contains all CranioZ projects.

        Each project lives in a subdirectory named after its
        ``ProjectId``.
        """
        if self._projects_root is None:
            raise AppPathInaccessible("resolver is not initialized")
        return self._projects_root

    @property
    def flows_root(self) -> Path:
        """Directory that contains the flow definitions."""
        if self._flows_root is None:
            raise AppPathInaccessible("resolver is not initialized")
        return self._flows_root

    @property
    def cache_root(self) -> Path:
        """Directory used for transient cached data."""
        if self._cache_root is None:
            raise AppPathInaccessible("resolver is not initialized")
        return self._cache_root

    # ----- Path resolution ------------------------------------------

    def resolve_in_projects_root(self, relative: str) -> Path:
        """Resolve a logical relative path against the projects root.

        The relative path is validated against traversal. No
        assumption is made about the existence of the target.

        This method is intended for low-level infrastructure code.
        Application code should use ``ProjectPathResolver`` for
        per-project paths.

        Raises:
            AppPathInaccessible: if the path escapes the root.
        """
        return self._resolve_under_root(self.projects_root, relative)

    # ----- Internals ------------------------------------------------

    @staticmethod
    def _resolve_under_root(root: Path, relative: str) -> Path:
        candidate = (root / relative).resolve()
        resolved_root = root.resolve()
        if resolved_root not in candidate.parents and candidate != resolved_root:
            raise AppPathInaccessible(f"path escapes root: {relative}")
        return candidate

    @staticmethod
    def _resolve_dir(
        *,
        configured: str | None,
        default: Path,
        field_name: str,
        create: bool,
    ) -> Path:
        path = Path(configured).expanduser() if configured else default

        if not path.exists():
            if not create:
                raise AppPathInaccessible(f"{field_name} does not exist: {path}")
            try:
                path.mkdir(parents=True, exist_ok=True)
            except OSError as exc:
                raise AppPathInaccessible(f"could not create {field_name}: {path}: {exc}") from exc

        if not path.is_dir():
            raise AppPathInaccessible(f"{field_name} is not a directory: {path}")
        return path

    # ----- Defaults -------------------------------------------------

    @staticmethod
    def _default_projects_root() -> Path:
        return Path.home() / "Documents" / DEFAULT_PROJECT_ROOT_NAME / DEFAULT_PROJECTS_DIR_NAME

    @staticmethod
    def _default_flows_root() -> Path:
        return Path.home() / "Documents" / DEFAULT_PROJECT_ROOT_NAME / DEFAULT_FLOWS_DIR_NAME

    @staticmethod
    def _default_cache_root() -> Path:
        return Path.home() / ".cranioz" / DEFAULT_CACHE_DIR_NAME

"""CatalogLoader.

Loads translation catalog files from disk.

A catalog is a JSON file organized by language and functional area,
for example:

    translations/en/core.json
    translations/pt-BR/workspace.json

This module is intentionally minimal:

- It reads and parses a single JSON file.
- It does not cache.
- It does not decide which catalogs to load.
- It does not know about the TranslationService or the registry.

Caching and decision-making belong to the TranslationCatalogRegistry.

The loader does not validate the catalog structure beyond what the
JSON parser requires. Semantic validation is the responsibility of
the TranslationValidator (a separate tool, not part of the runtime).
"""

from __future__ import annotations

import json
from pathlib import Path
from typing import Any


class CatalogLoaderError(Exception):
    """Raised when a catalog file cannot be read or parsed."""


class CatalogLoader:
    """Loads translation catalog files from a base directory.

    The base directory is the root of the translations tree, for
    example: src/cranioz/translations.
    """

    def __init__(self, base_dir: Path) -> None:
        if not base_dir.exists():
            raise CatalogLoaderError(f"catalog base directory not found: {base_dir}")
        if not base_dir.is_dir():
            raise CatalogLoaderError(f"catalog base path is not a directory: {base_dir}")
        self._base_dir = base_dir

    @property
    def base_dir(self) -> Path:
        """Return the base directory for translation catalogs."""
        return self._base_dir

    def load(self, language: str, area: str) -> dict[str, Any]:
        """Load a catalog for the given language and functional area.

        Args:
            language: BCP 47 language identifier (e.g. "en", "pt-BR").
            area: functional area (e.g. "core", "workspace").

        Returns:
            The parsed JSON contents as a dict.

        Raises:
            CatalogLoaderError: if the file is missing, unreadable,
                malformed, or does not contain a JSON object.
        """
        if not language:
            raise CatalogLoaderError("language must not be empty")
        if not area:
            raise CatalogLoaderError("area must not be empty")

        path = self._base_dir / language / f"{area}.json"

        if not path.exists():
            raise CatalogLoaderError(f"catalog file not found: {path}")

        try:
            with path.open("r", encoding="utf-8") as fh:
                data = json.load(fh)
        except json.JSONDecodeError as exc:
            raise CatalogLoaderError(f"invalid JSON in {path}: {exc}") from exc
        except OSError as exc:
            raise CatalogLoaderError(f"cannot read {path}: {exc}") from exc

        if not isinstance(data, dict):
            raise CatalogLoaderError(f"catalog must be a JSON object: {path}")
        return data

"""TranslationCatalogRegistry.

Locates, loads, and caches translation catalogs by language and
functional area.

Design rules:

- Catalogs are loaded on demand. Nothing is loaded at construction.
- Loaded catalogs are cached in memory, keyed by (language, area).
- A failed load is not cached: the next request will retry.
- The registry does not decide the active language. That is the
  TranslationService's responsibility.
- The registry does not merge catalogs. It returns the catalog for
  a single (language, area) pair. The TranslationService decides how
  to resolve a fully qualified key across areas.
- The registry supports explicit cache invalidation for development
  and testing.

The registry is a thin layer over the CatalogLoader. Its only added
value is caching and the (language, area) key.
"""

from __future__ import annotations

from typing import Any

from cranioz.infrastructure.i18n.catalog_loader import (
    CatalogLoader,
    CatalogLoaderError,
)


class CatalogRegistryError(Exception):
    """Raised when a catalog cannot be resolved."""


class TranslationCatalogRegistry:
    """Caches translation catalogs by (language, area)."""

    def __init__(self, loader: CatalogLoader) -> None:
        self._loader = loader
        self._cache: dict[tuple[str, str], dict[str, Any]] = {}

    # ----- Public API -----------------------------------------------

    def get(self, language: str, area: str) -> dict[str, Any]:
        """Return the catalog for (language, area).

        Loads the catalog on first request and caches it. Subsequent
        requests for the same pair return the cached instance.

        Raises:
            CatalogRegistryError: if the catalog cannot be loaded.
        """
        key = (language, area)
        cached = self._cache.get(key)
        if cached is not None:
            return cached

        try:
            catalog = self._loader.load(language, area)
        except CatalogLoaderError as exc:
            raise CatalogRegistryError(
                f"cannot load catalog for language={language!r} area={area!r}: {exc}"
            ) from exc

        self._cache[key] = catalog
        return catalog

    def has(self, language: str, area: str) -> bool:
        """Return True if the catalog is currently cached."""
        return (language, area) in self._cache

    def invalidate(self, language: str, area: str) -> None:
        """Remove a single entry from the cache.

        No-op if the entry is not cached.
        """
        self._cache.pop((language, area), None)

    def clear(self) -> None:
        """Remove all cached catalogs.

        Used during development and testing, or when the underlying
        catalog files change on disk and need to be reloaded.
        """
        self._cache.clear()

    def cached_keys(self) -> tuple[tuple[str, str], ...]:
        """Return the (language, area) pairs currently cached."""
        return tuple(self._cache.keys())

"""Global TranslationService instance (controlled singleton).

Widgets and UI code call ``get_translation_service()`` directly,
without receiving the service through a constructor. This avoids
threading the service through every widget's __init__.

To keep tests deterministic, the singleton is *controlled*:

- ``configure(service)`` replaces the current instance.
- ``reset()`` restores the default (an IdentityTranslationService).
- A test can install a controlled service in a fixture and reset
  it at teardown.

The module does not do any I/O at import time. The default instance
is created lazily on first access.
"""

from __future__ import annotations

from cranioz.infrastructure.i18n.translation_service import (

    TranslationService,
)

_instance: TranslationService | None = None


def get_translation_service() -> TranslationService:
    """Return the current global TranslationService.

    If no service has been configured yet, a default
    IdentityTranslationService is created and stored. The default
    returns the source text unchanged, which is safe: it never
    returns an internal key.
    """
    global _instance  # noqa: PLW0603
    if _instance is None:
        _instance = IdentityTranslationService()
    return _instance


def configure(service: TranslationService) -> None:
    """Install a new global TranslationService.

    Typically called once at application boot, after the catalogs
    have been discovered. It may also be called from tests to install
    a service backed by a controlled registry.

    Replaces the previous instance. The previous instance is not
    disposed: the TranslationService contract has no dispose method.

    global _instance  # noqa: PLW0603
    _instance = service


def reset() -> None:
    """Restore the default (IdentityTranslationService).

    Intended for tests: a fixture may call reset() at teardown so
    that subsequent tests do not see a leaked service.
    """
    global _instance  # noqa: PLW0603
    _instance = None

"""TranslationService.

Resolves translation keys to localized strings.

This module defines:

- The ``TranslationService`` contract, as a Protocol.
- A concrete implementation, ``CatalogTranslationService``, backed
  by a TranslationCatalogRegistry.
- An ``IdentityTranslationService``, useful for tests and for the
  early phase of the application before catalogs exist.

The public API is a single method::

    service.tr(key, source_text, **params)

The ``source_text`` argument is mandatory (see the i18n documentation
§5). It provides a last-resort fallback when the key cannot be
resolved through the catalogs.

Fallback sequence (i18n doc §8):

1. Look up the message in the active language catalog.
2. If unavailable, look up the message in the English source catalog.
3. If unavailable, use the required English ``source_text``.
4. If no usable text can be produced, use a safe generic message.

Today the service supports simple named-parameter interpolation
("Hello {name}"). ICU MessageFormat (pluralization, select, etc.) is
intentionally not implemented yet: no message in the codebase
requires it. It will be added when the first plural message appears.
"""

from __future__ import annotations

import logging
from typing import Any, Protocol, runtime_checkable

from cranioz.infrastructure.i18n.catalog_registry import (
    CatalogRegistryError,
    TranslationCatalogRegistry,
)

_logger = logging.getLogger(__name__)

SOURCE_LANGUAGE = "en"

# A safe generic message used as a last-resort fallback (step 4 of
# the fallback sequence). It is intentionally not translated: if the
# translation system cannot produce text, we must not depend on it.
_GENERIC_FALLBACK = "[translation unavailable]"


@runtime_checkable
class TranslationService(Protocol):
    """Contract for a translation service.

    Implementations must be able to:
    - return the active language;
    - change the active language;
    - resolve a translation key with a mandatory English source text.
    """

    @property
    def language(self) -> str: ...

    def set_language(self, language: str) -> None: ...

    def tr(self, key: str, source_text: str, **params: Any) -> str: ...


class CatalogTranslationService:
    """TranslationService backed by a TranslationCatalogRegistry.

    Keys are namespaced by functional area:

        "core.app.name"       → area "core",     path ["app", "name"]
        "workspace.area.x"    → area "workspace", path ["area", "x"]

    The first segment of the key selects the catalog. The remaining
    segments navigate the catalog tree to find an entry with a
    "message" field.

    If the key cannot be resolved through the active language catalog,
    the English catalog is tried. If that fails too, the required
    ``source_text`` is used. If even that is empty, a generic fallback
    message is returned and a diagnostic is logged.
    """

    def __init__(
        self,
        registry: TranslationCatalogRegistry,
        language: str = SOURCE_LANGUAGE,
    ) -> None:
        if not language:
            raise ValueError("language must not be empty")
        self._registry = registry
        self._language = language

    # ----- Language management --------------------------------------

    @property
    def language(self) -> str:
        return self._language

    def set_language(self, language: str) -> None:
        """Change the active language.

        No catalogs are loaded here: they are loaded on demand by the
        registry. Changing the language is therefore cheap.

        Raises ValueError if the language is empty.
        """
        if not language:
            raise ValueError("language must not be empty")
        self._language = language

    # ----- Translation ----------------------------------------------

    def tr(self, key: str, source_text: str, **params: Any) -> str:
        """Resolve a translation key.

        Args:
            key: semantic translation key, for example
                "workspace.area.placeholder.empty".
            source_text: mandatory English source text. Used as the
                last-resort fallback when the key is not resolvable
                through the catalogs.
            **params: named parameters for interpolation.

        Returns:
            The localized string, with parameters interpolated.
        """
        if not key:
            raise ValueError("key must not be empty")

        message = self._resolve(key, source_text)

        try:
            return message.format(**params)
        except (KeyError, IndexError, ValueError) as exc:
            _logger.warning("translation interpolation failed for key=%r: %s", key, exc)
            # Interpolation failure is a fallback event. Return the
            # source text unformatted so the user still sees something.
            return source_text or _GENERIC_FALLBACK

    # ----- Internals ------------------------------------------------

    def _resolve(self, key: str, source_text: str) -> str:
        """Apply the fallback sequence to resolve a key to a message."""
        message = self._lookup(self._language, key)
        if message is not None:
            return message

        if self._language != SOURCE_LANGUAGE:
            message = self._lookup(SOURCE_LANGUAGE, key)
            if message is not None:
                return message

        if source_text:
            return source_text

        _logger.error(
            "no usable translation for key=%r (language=%r)",
            key,
            self._language,
        )
        return _GENERIC_FALLBACK

    def _lookup(self, language: str, key: str) -> str | None:
        """Look up a key in the catalog of the given language.

        Returns the message string, or None if the catalog or key is
        unavailable.
        """
        area, *path = key.split(".")
        if not area or not path:
            return None

        try:
            catalog = self._registry.get(language, area)
        except CatalogRegistryError:
            return None

        node: Any = catalog
        for segment in path:
            if not isinstance(node, dict):
                return None
            node = node.get(segment)
            if node is None:
                return None

        if isinstance(node, dict):
            message = node.get("message")
            if isinstance(message, str):
                return message

        return None


class IdentityTranslationService:
    """TranslationService that returns the source_text unchanged.

    Useful for:
    - Tests that do not care about translations.
    - The early phase of the application before catalogs exist.
    - Fallback when no catalog registry is available.

    It supports the language property and set_language() so that
    callers see a consistent API, but changing the language has no
    effect on the returned text.
    """

    def __init__(self, language: str = SOURCE_LANGUAGE) -> None:
        if not language:
            raise ValueError("language must not be empty")
        self._language = language

    @property
    def language(self) -> str:
        return self._language

    def set_language(self, language: str) -> None:
        if not language:
            raise ValueError("language must not be empty")
        self._language = language

    def tr(self, key: str, source_text: str, **params: Any) -> str:
        if not key:
            raise ValueError("key must not be empty")
        try:
            return source_text.format(**params)
        except (KeyError, IndexError, ValueError):
            return source_text or _GENERIC_FALLBACK

"""ProjectPathResolver.

Resolves file paths inside a specific CranioZ project.

The resolver is initialized with two pieces of information:

- The projects root, provided by ``AppPathResolver.projects_root``.
- The ``ProjectId`` of the project being opened.

From those, it derives every path that belongs to the project:

    <projects_root>/<project_id>/
    <projects_root>/<project_id>/project.json
    <projects_root>/<project_id>/patient_record.json
    <projects_root>/<project_id>/workflow.json
    <projects_root>/<project_id>/imaging/ct.dcm
    <projects_root>/<project_id>/models/maxilla.stl
    ...

The resolver does NOT touch the filesystem beyond checking whether
the project directory exists (``exists()``). It does not create
directories, does not read files, and does not write anything. Those
responsibilities belong to the ProjectRepository.

Rules:

- The ``ProjectId`` value must be a safe path component. Values
  containing ``/``, ``\\``, ``..``, ``:``, control characters, or
  starting with ``.`` are rejected at construction time.
- Every path returned by this resolver is guaranteed to be inside
  the project directory. Traversal attempts through ``resource()``
  raise ``ProjectPathInaccessible``.
- The resolver is stateless after construction. It can be reused
  for the lifetime of the project.

See also:

- ``infrastructure/app/app_path_resolver.py`` — application-level
  paths (projects root, flows root, cache root).
- ADR-0004 and ``project_pt.md`` §14 — the design rationale for
  separating application paths from project paths.
"""

from __future__ import annotations

from pathlib import Path

from cranioz.domain.project.project_id import ProjectId

# Standard file names inside a project directory. Kept as module
# constants so they can be referenced in tests and in the
# repository without risk of typo.
PROJECT_FILE_NAME = "project.json"
PATIENT_RECORD_FILE_NAME = "patient_record.json"
WORKFLOW_FILE_NAME = "workflow.json"


class ProjectPathInaccessible(Exception):
    """Raised when a project path cannot be resolved."""


class ProjectPathResolver:
    """Resolves file paths inside a specific CranioZ project.

    The resolver is built once per project, typically when the
    ProjectManager opens the project. It does not touch the disk
    except when ``exists()`` is called.
    """

    def __init__(
        self,
        projects_root: Path,
        project_id: ProjectId,
    ) -> None:
        self._validate_project_id(project_id)
        self._projects_root = projects_root
        self._project_id = project_id

    # ----- Identity ------------------------------------------------

    @property
    def projects_root(self) -> Path:
        """The root directory that contains all projects."""
        return self._projects_root

    @property
    def project_id(self) -> ProjectId:
        """The ProjectId this resolver was built for."""
        return self._project_id

    # ----- Standard paths ------------------------------------------

    @property
    def project_dir(self) -> Path:
        """Directory of this project: ``<projects_root>/<project_id>``."""
        return self._projects_root / self._project_id.value

    @property
    def project_file(self) -> Path:
        """The project.json file of this project."""
        return self.project_dir / PROJECT_FILE_NAME

    @property
    def patient_record_file(self) -> Path:
        """The patient_record.json file of this project."""
        return self.project_dir / PATIENT_RECORD_FILE_NAME

    @property
    def workflow_file(self) -> Path:
        """The workflow.json file of this project."""
        return self.project_dir / WORKFLOW_FILE_NAME

    # ----- Free-form resources -------------------------------------

    def resource(self, relative: str) -> Path:
        """Resolve a logical relative path inside the project.

        Accepts POSIX-style (``"imaging/ct.dcm"``) and Windows-style
        (``"imaging\\ct.dcm"``) separators. Both are normalized to
        the platform's separator internally.

        The returned path is guaranteed to be inside the project
        directory. Traversal attempts (``..``) or absolute paths
        raise ``ProjectPathInaccessible``.

        The method does not check whether the target exists.
        """
        if not relative:
            raise ProjectPathInaccessible("relative path must not be empty")

        # Reject absolute paths in a platform-independent way.
        # Path(...).is_absolute() only detects platform-native
        # absolutes: on Windows it does not detect "/etc/passwd",
        # because a leading "/" is interpreted as the root of the
        # current drive. Since the resolver must behave the same on
        # every platform, we reject any leading separator and any
        # drive letter explicitly.
        if relative.startswith(("/", "\\")):
            raise ProjectPathInaccessible(f"relative path must not be absolute: {relative!r}")
        if len(relative) >= 2 and relative[1] == ":":
            # Windows drive letter, e.g. "C:...".
            raise ProjectPathInaccessible(f"relative path must not be absolute: {relative!r}")
        if Path(relative).is_absolute():
            # Covers any remaining platform-specific absolute form.
            raise ProjectPathInaccessible(f"relative path must not be absolute: {relative!r}")

        project_dir = self.project_dir.resolve()
        candidate = (project_dir / relative).resolve()

        if project_dir not in candidate.parents and candidate != project_dir:
            raise ProjectPathInaccessible(f"path escapes project directory: {relative!r}")
        return candidate

    # ----- State ---------------------------------------------------

    def exists(self) -> bool:
        """Return True if the project directory exists on disk.

        Does not check for individual files: the caller may want to
        inspect the project directory even if some files are missing.
        """
        return self.project_dir.is_dir()

    # ----- Validation ----------------------------------------------

    @staticmethod
    def _validate_project_id(project_id: ProjectId) -> None:
        """Reject ProjectIds that cannot be used as a path component.

        The ProjectId is a domain-level concept; it may contain any
        string the domain allows. This resolver, however, is about
        to embed it into a filesystem path, so it must be a safe
        single path segment.

        Rejected patterns:

        - empty or whitespace-only
        - contains ``/`` or ``\\`` (path separators)
        - contains ``..`` (traversal)
        - contains ``:`` (Windows drive separator, NTFS stream
          separator)
        - contains ASCII control characters (0x00-0x1F)
        - starts with ``.`` (would produce hidden directories, and
          would allow ``.`` or ``..`` themselves)
        """
        value = project_id.value

        if not value or not value.strip():
            raise ProjectPathInaccessible("ProjectId value must not be empty or whitespace only")
        if "/" in value or "\\" in value:
            raise ProjectPathInaccessible(
                f"ProjectId value must not contain path separators: {value!r}"
            )
        if ".." in value:
            raise ProjectPathInaccessible(f"ProjectId value must not contain '..': {value!r}")
        if ":" in value:
            raise ProjectPathInaccessible(f"ProjectId value must not contain ':': {value!r}")
        if any(ord(ch) < 0x20 for ch in value):
            raise ProjectPathInaccessible(
                f"ProjectId value must not contain control characters: {value!r}"
            )
        if value.startswith("."):
            raise ProjectPathInaccessible(f"ProjectId value must not start with '.': {value!r}")

"""Editor ABC.

An Editor is a functional unit hosted by an Area. It provides a specific
interaction experience. It does not know where it is placed; the Area
hosts it.

This module defines the contract only. Concrete Editors may use Qt for
materialization, but the contract itself has no Qt dependency.
"""

from abc import ABC, abstractmethod
from typing import TYPE_CHECKING, Any

if TYPE_CHECKING:
    from cranioz.ui.editors.editor_context import EditorContext
    from cranioz.ui.toolbars.specification import ToolbarSpec


class Editor(ABC):
    """Abstract Editor contract."""

    editor_id: str
    title: str

    def __init__(self, editor_id: str, title: str) -> None:
        self.editor_id = editor_id
        self.title = title

    # Lifecycle -----------------------------------------------------

    def initialize(self, context: EditorContext) -> None:
        """Called once at insertion. Heavy resources are set up here."""
        return None

    def activate(self) -> None:
        """Called when the Editor becomes the active one in its Area."""
        return None

    def deactivate(self) -> None:
        """Called when the Editor stops being active. Must not dispose."""
        return None

    def dispose(self) -> None:
        """Called when the Editor is removed and its resources released."""
        return None

    # Content -------------------------------------------------------

    @abstractmethod
    def content_widget(self) -> Any:
        """Return the widget that represents the Editor's content."""
        raise NotImplementedError

    def toolbar_specs(self) -> list[ToolbarSpec]:
        """Declare the Editor's internal Toolbars. Empty by default."""
        return []

    # Persistence ---------------------------------------------------

    def persistent_state(self) -> dict[str, Any]:
        """Return the Editor's persistable presentation state."""
        return {}

    def restore_state(self, state: dict[str, Any]) -> None:
        """Restore a previously saved presentation state."""
        return None

"""EditorRegistry.

Registers Editor classes by identifier and creates instances.

Rules:
- The Registry registers types, not instances.
- The Registry resolves editor_id to a class (type[Editor]).
- Creating an instance delegates to the Editor's own constructor;
  the caller supplies the editor_id, title, and any extra keyword
  arguments the concrete Editor requires.
- Availability and enablement are not decided here; they are the
  responsibility of the EditorCatalog, which filters registered
  types by context.
- Re-registering the same identifier with the same class is
  idempotent; with a different class it raises ValueError.
- Resolving or creating an unknown identifier raises KeyError.
"""

from __future__ import annotations

from typing import Any

from cranioz.ui.editors.base_editor import Editor


class EditorRegistryError(ValueError):
    """Raised when registration violates a structural rule."""


class EditorRegistry:
    """Registry of Editor classes by identifier."""

    def __init__(self) -> None:
        self._types: dict[str, type[Editor]] = {}

    def register(self, editor_id: str, editor_type: type[Editor]) -> None:
        """Register an Editor class under editor_id.

        Idempotent when the same class is registered again.
        Raises EditorRegistryError when a different class is already
        registered under the same identifier, or when editor_id is
        empty.
        """
        if not editor_id:
            raise EditorRegistryError("editor_id must not be empty")
        if editor_id in self._types:
            if self._types[editor_id] is editor_type:
                return
            raise EditorRegistryError(
                f"editor_id already registered with a different type: {editor_id}"
            )
        self._types[editor_id] = editor_type

    def unregister(self, editor_id: str) -> None:
        """Remove a class. No-op if the identifier is not registered."""
        self._types.pop(editor_id, None)

    def resolve(self, editor_id: str) -> type[Editor]:
        """Return the class registered for editor_id.

        Raises KeyError if unknown.
        """
        try:
            return self._types[editor_id]
        except KeyError as exc:
            raise KeyError(f"editor_id not registered: {editor_id}") from exc

    def create(self, id_: str, **kwargs: Any) -> Editor:
        """Create a new Editor instance for the given editor id.

        The first parameter is named `id_` to avoid colliding with a
        keyword argument named `editor_id` that concrete Editors may
        accept in their constructors.

        The caller supplies keyword arguments accepted by the concrete
        Editor constructor. If none are given, the Editor must be
        constructible with no arguments.

        Raises KeyError if unknown.
        """
        editor_type = self.resolve(id_)
        return editor_type(**kwargs)

    def identifiers(self) -> tuple[str, ...]:
        """Return the registered identifiers in registration order."""
        return tuple(self._types.keys())

    def contains(self, editor_id: str) -> bool:
        """Return True if the identifier is registered."""
        return editor_id in self._types

"""ThemeService.

Provides theme-dependent resources to UI widgets.

In this phase the service is minimal: it exposes the active theme name
and resolves logical icon names into concrete icon identifiers.

Concrete visual styling (colors, fonts, spacing) is applied later via
Qt stylesheets. Widgets should never hardcode colors or icon paths;
they consult the ThemeService instead.

Rules:
- The service is stateless with respect to the application; it only
  reports the active theme and resolves icons.
- The service does not decide *when* the theme changes; that is the
  responsibility of a future settings/theme coordinator.
- The service does not know about individual widgets.
"""

from __future__ import annotations

from dataclasses import dataclass, replace
from enum import Enum


class ThemeName(Enum):
    """Identifier of the active theme."""

    LIGHT = "light"
    DARK = "dark"


@dataclass(frozen=True)
class ThemeService:
    """Resolves theme-dependent resources for UI widgets.

    The service is immutable. To change theme, create a new instance
    via `with_theme`.
    """

    theme: ThemeName = ThemeName.DARK

    def with_theme(self, theme: ThemeName) -> ThemeService:
        """Return a new ThemeService for the given theme."""
        return replace(self, theme=theme)

    def icon_name(self, logical_name: str) -> str:
        """Return the concrete icon name for the active theme.

        For now this is identity: the logical name is the icon name.
        Future themes may map 'log' to 'log_dark' or 'log_light'.

        Rules:
        - The logical name is what Tools declare in their specs.
        - The concrete name is what the UI loader will look up on disk.
        - The mapping depends only on the active theme.
        """
        if not logical_name:
            raise ValueError("logical_name must not be empty")
        return logical_name

"""Toolbar widget.

Materializes a ToolbarSpec as a Qt widget. It is internal to an Editor;
the Workspace never hosts one.

Responsibilities:
- Resolve each tool_id declared in the spec against the ToolRegistry.
- Create the concrete Tool instances for each item.
- Display the items in the declared orientation.
- Consume a ThemeService for icon resolution.
- Emit a signal when a Tool is triggered or toggled.

Non-responsibilities:
- The Toolbar does not decide availability or enablement.
  The Tool decides those, based on a ToolContext that the Editor
  provides at materialization time.
- The Toolbar does not mutate the Scene. Tools do, via Commands.
- The Toolbar does not persist state. That is the Editor's job.

This widget is deliberately minimal. It is the first Qt widget in
CranioZ and it exists to validate the contract between:
  ToolbarSpec + ToolRegistry + ThemeService → QWidget
"""

from __future__ import annotations

from dataclasses import dataclass

from PySide6.QtCore import Qt, Signal
from PySide6.QtWidgets import (
    QHBoxLayout,
    QPushButton,
    QVBoxLayout,
    QWidget,
)

from cranioz.ui.theme.service import ThemeService
from cranioz.ui.toolbars.specification import (
    ToolbarOrientation,
    ToolbarSpec,
)
from cranioz.ui.tools.registry.tool_definition import Tool, ToolContext
from cranioz.ui.tools.registry.tool_registry import ToolRegistry


@dataclass(frozen=True)
class ToolbarItemBinding:
    """A materialized item, linking a ToolbarItem to a Tool instance."""

    tool_id: str
    widget: QWidget
    tool: Tool | None


class Toolbar(QWidget):
    """Qt widget materializing a ToolbarSpec."""

    tool_triggered = Signal(str)  # tool_id
    tool_toggled = Signal(str, bool)  # tool_id, checked

    def __init__(
        self,
        spec: ToolbarSpec,
        registry: ToolRegistry,
        theme: ThemeService,
        context: ToolContext | None = None,
        parent: QWidget | None = None,
    ) -> None:
        super().__init__(parent)
        self._spec = spec
        self._registry = registry
        self._theme = theme
        self._context = context if context is not None else ToolContext()
        self._bindings: list[ToolbarItemBinding] = []

        self.setObjectName(f"toolbar.{spec.toolbar_id}")
        self.setWindowTitle(spec.title)

        self._layout = (
            QHBoxLayout(self)
            if spec.orientation is ToolbarOrientation.HORIZONTAL
            else QVBoxLayout(self)
        )
        self._layout.setContentsMargins(4, 4, 4, 4)
        self._layout.setSpacing(4)
        self._layout.setAlignment(Qt.AlignmentFlag.AlignLeft)

        for tool_id in spec.tool_ids:
            self._append_tool(tool_id)

    # ----- Public API ------------------------------------------------

    @property
    def spec(self) -> ToolbarSpec:
        return self._spec

    def bindings(self) -> tuple[ToolbarItemBinding, ...]:
        """Return the materialized item bindings, in declaration order."""
        return tuple(self._bindings)

    def tool_ids(self) -> tuple[str, ...]:
        return tuple(b.tool_id for b in self._bindings)

    # ----- Internals -------------------------------------------------

    def _append_tool(self, tool_id: str) -> None:
        tool = self._registry.create(tool_id)
        label = tool.label or tool_id
        icon_name = self._theme.icon_name(tool.icon) if tool.icon else None

        # For this first version we render Tools as buttons with their label.
        # Icon resolution is exercised but not yet drawn (no icon loader).
        button = QPushButton(label, self)
        button.setObjectName(f"toolbar.item.{tool_id}")
        button.setToolTip(tool.description or label)
        if icon_name is not None:
            button.setProperty("icon_name", icon_name)

        if self._is_toggle(tool):
            button.setCheckable(True)
            button.toggled.connect(lambda checked, tid=tool_id: self._on_toggle(tid, checked))
        else:
            button.clicked.connect(lambda _=False, tid=tool_id: self._on_click(tid))

        self._layout.addWidget(button)
        self._bindings.append(ToolbarItemBinding(tool_id=tool_id, widget=button, tool=tool))

    def _is_toggle(self, tool: Tool) -> bool:
        return hasattr(tool, "toggle") and hasattr(tool, "checked")

    def _on_click(self, tool_id: str) -> None:
        binding = self._binding_for(tool_id)
        tool = binding.tool
        if tool is None:
            return
        if not tool.is_available(self._context) or not tool.is_enabled(self._context):
            return
        execute = getattr(tool, "execute", None)
        if callable(execute):
            execute(self._context)
        self.tool_triggered.emit(tool_id)

    def _on_toggle(self, tool_id: str, checked: bool) -> None:
        binding = self._binding_for(tool_id)
        tool = binding.tool
        if tool is None:
            return
        toggle = getattr(tool, "toggle", None)
        if callable(toggle):
            current = toggle()
            # The tool's new state wins.
            checked = bool(current)
        self.tool_toggled.emit(tool_id, checked)

    def _binding_for(self, tool_id: str) -> ToolbarItemBinding:
        for binding in self._bindings:
            if binding.tool_id == tool_id:
                return binding
        raise KeyError(f"tool_id not materialized in this toolbar: {tool_id}")

"""ToolbarSpec.

Declarative description of a Toolbar inside an Editor.

A Toolbar is internal to an Editor; the Workspace never hosts one.
The same Tool may be referenced by multiple Toolbars across Editors.

Two concepts are separated here:

- ToolbarSpec describes *what* a Toolbar is: identity, orientation,
  tools, and its presentation policy.
- ToolbarState (see state.py) describes *whether* a Toolbar is
  currently visible. It is runtime state, not declarative.

Presentation policy is declarative and does not change at runtime.
Runtime visibility is state and does change at runtime.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from enum import Enum


class ToolbarOrientation(Enum):
    """Orientation of a Toolbar."""

    HORIZONTAL = "horizontal"
    VERTICAL = "vertical"


class ToolbarVisibilityMode(Enum):
    """Presentation policy for a Toolbar.

    ALWAYS
        Always visible. The user cannot hide it.

    COLLAPSIBLE
        Visible by default. The user may collapse or expand it
        manually. The preference is persisted across sessions.

    AUTO_HIDE
        Hidden by default. The Editor reveals it when the mouse
        approaches its edge zone (hover-reveal). It hides again
        when the mouse leaves the zone. The user does not toggle
        it manually.

    HIDDEN
        Not visible by default. It can only be revealed by an
        explicit action (a menu, a shortcut, a command).
    """

    ALWAYS = "always"
    COLLAPSIBLE = "collapsible"
    AUTO_HIDE = "auto_hide"
    HIDDEN = "hidden"


class ToolbarSpecError(ValueError):
    """Raised when a ToolbarSpec violates structural rules."""


@dataclass(frozen=True)
class ToolbarSpec:
    """Declarative description of a Toolbar."""

    toolbar_id: str
    title: str
    orientation: ToolbarOrientation = ToolbarOrientation.HORIZONTAL
    tool_ids: tuple[str, ...] = field(default_factory=tuple)
    visibility_mode: ToolbarVisibilityMode = ToolbarVisibilityMode.ALWAYS

    def __post_init__(self) -> None:
        if not self.toolbar_id:
            raise ToolbarSpecError("toolbar_id must not be empty")
        if not self.title:
            raise ToolbarSpecError("title must not be empty")

"""ToolbarState.

Runtime visibility state of a Toolbar.

A ToolbarSpec declares *how* a Toolbar should behave (its visibility
policy). A ToolbarState records *whether* it is currently visible.

State is per-instance, per-session. It may be persisted via
Editor.persistent_state() when the policy is COLLAPSIBLE.

"""ToolRegistry.

Registers Tool implementations by identifier and creates instances.

Rules:
- The Registry registers types, not instances.
- The Registry resolves tool_id to a factory (a zero-argument callable).
- The Registry does not decide availability or enablement; that is
  the responsibility of the Tool itself via is_available and is_enabled,
  given a ToolContext.
- Re-registering the same identifier raises ValueError unless the same
  type is registered again (idempotent) — this avoids silent overrides.
- Resolving or creating an unknown identifier raises KeyError.
"""

from __future__ import annotations

from collections.abc import Callable

from cranioz.ui.tools.registry.tool_definition import Tool


class ToolRegistryError(ValueError):
    """Raised when registration violates a structural rule."""


ToolFactory = Callable[[], Tool]


class ToolRegistry:
    """Registry of Tool factories by identifier."""

    def __init__(self) -> None:
        self._factories: dict[str, ToolFactory] = {}

    def register(self, tool_id: str, factory: ToolFactory) -> None:
        """Register a factory under tool_id.

        Idempotent for the same factory; raises if a different factory
        is already registered under the same identifier.
        """
        if not tool_id:
            raise ToolRegistryError("tool_id must not be empty")
        if tool_id in self._factories:
            if self._factories[tool_id] is factory:
                return
            raise ToolRegistryError(
                f"tool_id already registered with a different factory: {tool_id}"
            )
        self._factories[tool_id] = factory

    def unregister(self, tool_id: str) -> None:
        """Remove a factory. No-op if the identifier is not registered."""
        self._factories.pop(tool_id, None)

    def resolve(self, tool_id: str) -> ToolFactory:
        """Return the factory for tool_id.

        Raises KeyError if unknown.
        """
        try:
            return self._factories[tool_id]
        except KeyError as exc:
            raise KeyError(f"tool_id not registered: {tool_id}") from exc

    def create(self, tool_id: str) -> Tool:
        """Create a new Tool instance for tool_id.

        Raises KeyError if unknown.
        """
        return self.resolve(tool_id)()

    def identifiers(self) -> tuple[str, ...]:
        """Return the registered identifiers in registration order."""
        return tuple(self._factories.keys())

    def contains(self, tool_id: str) -> bool:
        """Return True if the identifier is registered."""
        return tool_id in self._factories

"""Workspace.

The Workspace is the runtime environment where clinical work happens.
It is the composition of Areas — containers that host Editors — and
Overlays — floating layers over Regions.

This class is the *facade* of the Workspace subsystem. It ties
together the lower-level components built in earlier pieces:

- LayoutBuilder: turns a ModuleUISpec into domain structures
  (Areas, LayoutTree, populated AreaManager, Editors).
- LayoutRenderer: materializes the LayoutTree into Qt widgets
  (nested QSplitters and AreaWidgets).
- OverlayManager: registers and repositions Overlays by Placement.

Public responsibilities:

- ``load_spec(spec)``: materialize a ModuleUISpec. Rebuilds the
  whole visual tree and repositions every registered Overlay.
- ``clear()``: drop the current composition (Areas, widgets,
  Overlays registrations remain; they can be re-added on the next
  ``load_spec``).
- ``areas()``: return the Areas of the current composition.
- ``overlay_manager``: read-only access for advanced callers.
- ``top_overlay``: the shared Top Bar Overlay, exposed so the
  Workspace host can populate its content.

Non-responsibilities (future pieces):

- Persistence of layout state (Bloco 11).
- Drag-and-drop, floating Areas, multi-monitor (Bloco 14).
- Top Bar content: the three segments are not built here; they
  will be added by a dedicated Top Bar composition.
- Steps Panel: it is a second shared Overlay, to be added when the
  Flow system is implemented (Bloco 9).

Design rules:

- The Workspace is a QWidget. Its single visual child is the root
  widget returned by the LayoutRenderer.
- ``load_spec()`` is idempotent in the sense that calling it again
  replaces the previous composition. It always calls ``clear()``
  first.
- The Workspace owns the AreaManager produced by the build result.
  It does not copy it: the same manager is used throughout the
  lifecycle of the composition.
- The Workspace owns the LayoutRenderer instance. The renderer
  caches widget instances; the Workspace is responsible for
  clearing that cache when the composition is replaced.
- The TopOverlay is registered in the OverlayManager on every
  ``load_spec()``. Registration is idempotent, so re-registering
  the same instance on subsequent calls is safe.
"""

from __future__ import annotations

from PySide6.QtCore import QTimer
from PySide6.QtGui import QResizeEvent
from PySide6.QtWidgets import QVBoxLayout, QWidget

from cranioz.ui.editors.editor_registry import EditorRegistry
from cranioz.ui.theme.service import ThemeService
from cranioz.ui.tools.registry.tool_registry import ToolRegistry
from cranioz.ui.workspace.areas.area import Area
from cranioz.ui.workspace.areas.area_manager import AreaManager
from cranioz.ui.workspace.areas.placement import Placement
from cranioz.ui.workspace.layout.layout_builder import (
    BuildResult,
    LayoutBuilder,
)
from cranioz.ui.workspace.layout.layout_renderer import LayoutRenderer
from cranioz.ui.workspace.overlays.overlay_manager import OverlayManager
from cranioz.ui.workspace.overlays.region import Bounds, Region
from cranioz.ui.workspace.overlays.top_overlay import TopOverlay
from cranioz.ui.workspace.specs.module_ui_spec import ModuleUISpec


class WorkspaceError(RuntimeError):
    """Raised when the Workspace is used in an invalid state."""


class Workspace(QWidget):
    """The Workspace facade.

    Hosts the current composition, owns the LayoutRenderer and the
    OverlayManager, and exposes ``load_spec()`` as the single entry
    point for building a new layout.
    """

    def __init__(
        self,
        editor_registry: EditorRegistry,
        tool_registry: ToolRegistry,
        theme_service: ThemeService,
        debug_colors: bool = False,
        parent: QWidget | None = None,
    ) -> None:
        super().__init__(parent)
        self.setObjectName("workspace")

        self._editor_registry = editor_registry
        self._tool_registry = tool_registry
        self._theme_service = theme_service
        self._debug_colors = debug_colors

        self._layout_builder = LayoutBuilder(editor_registry)
        self._layout_renderer = LayoutRenderer(
            tool_registry,
            theme_service,
            debug_colors=debug_colors,
        )
        self._overlay_manager = OverlayManager()
        self._top_overlay = TopOverlay(parent=self)

        # Current composition. None until load_spec() is called.
        self._build_result: BuildResult | None = None
        self._root_widget: QWidget | None = None

        # The Workspace has a single child: the root widget produced
        # by the LayoutRenderer. A QVBoxLayout keeps it filling the
        # Workspace.
        self._root_layout = QVBoxLayout(self)
        self._root_layout.setContentsMargins(0, 0, 0, 0)
        self._root_layout.setSpacing(0)

        # The TopOverlay is registered from the start so that callers
        # can access it via ``top_overlay()`` even before the first
        # ``load_spec()``. Its widget is a child of the Workspace;
        # the OverlayManager does not add it to the layout.
        self._overlay_manager.register(self._top_overlay)
        self._top_overlay.widget().setParent(self)

        if debug_colors:
            # Neutral backdrop for the Workspace so Area colors
            # stand out. Also paints the TopOverlay so it becomes
            # visible during debug.
            self.setStyleSheet("QWidget#workspace { background-color: #ffffff; }")
            self._top_overlay.widget().setStyleSheet(
                "QFrame#top_overlay { background-color: #fadbd8; border: 1px solid #e74c3c; }"
            )

    # ----- Public API -----------------------------------------------

    def load_spec(self, spec: ModuleUISpec) -> None:
        """Materialize the given ModuleUISpec.

        Replaces any previous composition. After this call:

        - ``areas()`` returns the Areas of the new composition.
        - The visual tree reflects the new LayoutTree.
        - Every registered Overlay has been repositioned.
        """
        # Drop any previous composition.
        self.clear()

        # Build the domain structures.
        result = self._layout_builder.build(spec)
        self._build_result = result

        # Render the LayoutTree.
        root = self._layout_renderer.render(result.tree)
        self._root_widget = root
        self._root_layout.addWidget(root)

        # The TopOverlay must be a child of the Workspace so it can
        # float above the root widget. Reassert the parent here in
        # case the previous clear() detached it.
        self._top_overlay.widget().setParent(self)
        self._top_overlay.widget().raise_()

        # Position every Overlay according to the current geometry.
        # At this point the widgets may not yet have been laid out
        # by Qt; the bounds will be recomputed on the first resize.
        # If Qt has already laid them out, the bounds are accurate
        # immediately.
        self._reposition_overlays()

        # Also schedule a reposition after Qt has processed the
        # current event loop iteration. This ensures the overlays
        # are correctly positioned after the widget hierarchy has
        # been laid out by Qt, which may not have happened yet at
        # the moment load_spec() returns.
        QTimer.singleShot(0, self._reposition_overlays)

    def clear(self) -> None:
        """Remove the current composition.

        The Workspace keeps its registries and its TopOverlay. Only
        the rendered composition and the areas from the previous
        spec are removed.
        """
        # Detach and destroy the root widget and everything under it.
        if self._root_widget is not None:
            self._root_layout.removeWidget(self._root_widget)
            self._root_widget.setParent(None)
            self._root_widget.deleteLater()
            self._root_widget = None

        # Clear the renderer's widget cache so the next render()
        # creates fresh widgets.
        self._layout_renderer.clear_cache()

        # Forget the previous build result.
        self._build_result = None

    def areas(self) -> tuple[Area, ...]:
        """Return the Areas of the current composition.

        Returns an empty tuple when no spec has been loaded.
        """
        if self._build_result is None:
            return ()
        return self._build_result.areas

    def area_manager(self) -> AreaManager | None:
        """Return the AreaManager of the current composition.

        Returns None when no spec has been loaded.
        """
        if self._build_result is None:
            return None
        return self._build_result.area_manager

    def overlay_manager(self) -> OverlayManager:
        """Return the OverlayManager.

        The manager is stable across ``load_spec()`` calls: the
        Workspace owns a single instance.
        """
        return self._overlay_manager

    def top_overlay(self) -> TopOverlay:
        """Return the shared Top Bar Overlay."""
        return self._top_overlay

    def is_loaded(self) -> bool:
        """Return True if a composition has been loaded."""
        return self._build_result is not None

    # ----- Qt events ------------------------------------------------

    def resizeEvent(self, event: QResizeEvent) -> None:  # noqa: N802
        """Reposition Overlays when the Workspace is resized.

        The AreaWidgets are resized by the QSplitters they live in.
        As soon as the widgets' geometries change, the Region bounds
        change as well. This handler recomputes the bounds and asks
        the OverlayManager to reposition every Overlay.
        """
        super().resizeEvent(event)
        self._reposition_overlays()

    # ----- Internals ------------------------------------------------

    def _reposition_overlays(self) -> None:
        """Compute the current Region bounds and reposition Overlays."""
        bounds_by_placement = self._layout_renderer.compute_region_bounds()
        regions = self._regions_from_bounds(bounds_by_placement)
        self._overlay_manager.reposition_all(regions)

        # Ensure the TopOverlay stays above the composition. Qt
        # sometimes re-orders widgets when the layout is re-laid
        # out, and the composition's child widgets can end up above
        # the overlay. Raising the overlay after every reposition
        # keeps it on top.
        self._top_overlay.widget().raise_()

    def _regions_from_bounds(
        self,
        bounds_by_placement: dict[Placement, Bounds],
    ) -> tuple[Region, ...]:
        """Build Regions from the current AreaManager and bounds.

        Every Placement that has both Areas in the current
        composition and bounds computed by the renderer produces a
        Region. The Region carries the Areas from the AreaManager
        so that downstream consumers (Overlays, future layout I/O)
        can inspect which Areas belong to which Region.
        """
        manager = self.area_manager()
        if manager is None:
            return ()

        regions: list[Region] = []
        for placement, bounds in bounds_by_placement.items():
            areas = tuple(area for area in manager.areas() if area.placement is placement)
            if not areas:
                continue
            regions.append(
                Region(
                    region_id=placement.value,
                    bounds=bounds,
                    areas=areas,
                    semantic_role=placement,
                )
            )
        return tuple(regions)

"""Area.

An Area is a structural container that hosts Editors. It provides
the space where Editors are placed; it does not decide where it is
placed within the Workspace.

Rules:
- An Area has a stable id, a title, and a placement.
- The placement expresses the Area's semantic role in the Workspace
  (CENTRAL, LEFT, RIGHT, BOTTOM). It is a property of the Area,
  because the Area needs to know its role to be rendered correctly
  even when empty.
- The Area does not know where it is placed in the Workspace. Placement
  is a semantic role, not a position. Positions are decided by the
  LayoutTree.
- The Area does not know what its Editors do. It provides add/remove
  and activation, nothing more.
- The Area does not materialize Toolbars. Toolbars are internal to
  Editors (see ADR-0002).
- The Area does not know about Modules, Scenes, Commands, or the
  Workspace. It is a UI container.
- An Area may be empty (zero Editors). This is a valid state,
  especially for the CENTRAL Area, which must always exist as a
  container even when it hosts nothing.

This is a pure UI container. It has no Qt dependency at this level;
the widget materialization is the LayoutRenderer's job.
"""

from __future__ import annotations

from cranioz.ui.editors.base_editor import Editor
from cranioz.ui.workspace.areas.placement import Placement


class AreaError(ValueError):
    """Raised when an Area operation violates a structural rule."""


class Area:
    """A structural container that hosts Editors."""

    def __init__(
        self,
        area_id: str,
        title: str,
        placement: Placement,
    ) -> None:
        if not area_id:
            raise AreaError("area_id must not be empty")
        if not title:
            raise AreaError("title must not be empty")

        self._area_id = area_id
        self._title = title
        self._placement = placement
        self._editors: list[Editor] = []
        self._active_editor_id: str | None = None
        self._visible = True

    # ----- Identity ---------------------------------------------------

    @property
    def area_id(self) -> str:
        return self._area_id

    @property
    def title(self) -> str:
        return self._title

    @property
    def placement(self) -> Placement:
        return self._placement

    # ----- Visibility -------------------------------------------------

    @property
    def visible(self) -> bool:
        return self._visible

    def show(self) -> None:
        self._visible = True

    def hide(self) -> None:
        self._visible = False

    # ----- Editor management -----------------------------------------

    def add_editor(self, editor: Editor) -> None:
        """Add an Editor to this Area.

        Raises AreaError if an Editor with the same editor_id is
        already hosted by this Area.
        """
        for existing in self._editors:
            if existing.editor_id == editor.editor_id:
                raise AreaError(f"editor_id already hosted by this Area: {editor.editor_id}")
        self._editors.append(editor)
        if self._active_editor_id is None:
            self._active_editor_id = editor.editor_id

    def remove_editor(self, editor_id: str) -> Editor | None:
        """Remove an Editor by editor_id.

        Returns the removed Editor, or None if not found.
        If the removed Editor was active, the next hosted Editor
        becomes active. If no Editors remain, the active id is cleared.
        Removing the last Editor does NOT remove the Area. An empty
        Area is a valid state, especially for the CENTRAL Area.
        """
        for index, editor in enumerate(self._editors):
            if editor.editor_id == editor_id:
                removed = self._editors.pop(index)
                if self._active_editor_id == editor_id:
                    self._active_editor_id = self._editors[0].editor_id if self._editors else None
                return removed
        return None

    def editors(self) -> tuple[Editor, ...]:
        """Return the hosted Editors in insertion order."""
        return tuple(self._editors)

    def get_editor(self, editor_id: str) -> Editor | None:
        """Return the Editor with the given editor_id, or None."""
        for editor in self._editors:
            if editor.editor_id == editor_id:
                return editor
        return None

    def contains(self, editor_id: str) -> bool:
        """Return True if an Editor with the given id is hosted."""
        return any(e.editor_id == editor_id for e in self._editors)

    def editor_count(self) -> int:
        """Return the number of hosted Editors."""
        return len(self._editors)

    def is_empty(self) -> bool:
        """Return True if the Area hosts no Editors."""
        return not self._editors

    # ----- Active Editor ---------------------------------------------

    @property
    def active_editor_id(self) -> str | None:
        return self._active_editor_id

    @property
    def active_editor(self) -> Editor | None:
        if self._active_editor_id is None:
            return None
        return self.get_editor(self._active_editor_id)

    def set_active_editor(self, editor_id: str) -> None:
        """Set the active Editor by editor_id.

        Raises AreaError if the Editor is not hosted by this Area.
        """
        if not self.contains(editor_id):
            raise AreaError(f"editor_id is not hosted by this Area: {editor_id}")
        self._active_editor_id = editor_id

    # ----- Representation --------------------------------------------

    def __repr__(self) -> str:
        return (
            f"Area(area_id={self._area_id!r}, "
            f"title={self._title!r}, "
            f"placement={self._placement.value!r}, "
            f"visible={self._visible}, "
            f"editors={len(self._editors)}, "
            f"active={self._active_editor_id!r})"
        )

"""AreaManager.

Registry of Area instances within a Workspace.

Rules:
- The manager registers Area instances, not types.
- The manager preserves insertion order when listing Areas.
- Re-registering the same instance under the same area_id is
  idempotent. Registering a different instance with the same
  area_id raises ValueError.
- get() returns None for unknown ids. require() raises KeyError.
- The manager does not create Areas. Construction of Areas is the
  responsibility of the LayoutBuilder.
- The manager does not decide visibility, activation, or layout.
  It only tracks which Areas exist in this Workspace instance.
- The manager does not enforce the "at least one CENTRAL Area"
  invariant by itself, but it exposes helpers (count_by_placement,
  has_central) that the Workspace uses to enforce it.

The AreaManager is per-Workspace. Two Workspaces have two managers.
"""

from __future__ import annotations

from cranioz.ui.workspace.areas.area import Area
from cranioz.ui.workspace.areas.placement import Placement


class AreaManagerError(ValueError):
    """Raised when area registration violates a structural rule."""


class AreaManager:
    """Registry of Area instances."""

    def __init__(self) -> None:
        self._areas: dict[str, Area] = {}

    def register(self, area: Area) -> None:
        """Register an Area.

        Idempotent for the same instance. Raises AreaManagerError when
        a different Area is already registered under the same area_id.
        """
        area_id = area.area_id
        existing = self._areas.get(area_id)
        if existing is area:
            return
        if existing is not None:
            raise AreaManagerError(f"area_id already registered with a different Area: {area_id}")
        self._areas[area_id] = area

    def unregister(self, area_id: str) -> Area | None:
        """Remove an Area. Returns the removed Area, or None if unknown."""
        return self._areas.pop(area_id, None)

    def get(self, area_id: str) -> Area | None:
        """Return the Area with the given area_id, or None."""
        return self._areas.get(area_id)

    def require(self, area_id: str) -> Area:
        """Return the Area with the given area_id.

        Raises KeyError if unknown.
        """
        try:
            return self._areas[area_id]
        except KeyError as exc:
            raise KeyError(f"area_id not registered: {area_id}") from exc

    def contains(self, area_id: str) -> bool:
        """Return True if an Area with the given id is registered."""
        return area_id in self._areas

    def areas(self) -> tuple[Area, ...]:
        """Return the registered Areas in insertion order."""
        return tuple(self._areas.values())

    def area_count(self) -> int:
        """Return the number of registered Areas."""
        return len(self._areas)

    def count_by_placement(self, placement: Placement) -> int:
        """Return the number of registered Areas with the given placement."""
        return sum(1 for area in self._areas.values() if area.placement is placement)

    def has_central(self) -> bool:
        """Return True if at least one CENTRAL Area is registered."""
        return self.count_by_placement(Placement.CENTRAL) > 0

    def clear(self) -> None:
        """Remove all registered Areas.

        Used during Workspace shutdown and during module switches.
        """
        self._areas.clear()

    def __repr__(self) -> str:
        return f"AreaManager(areas={list(self._areas.keys())!r})"

"""AreaWidget.

Qt widget that materializes a single Area.

The Area is a pure domain model (no Qt). The AreaWidget is the
presentation layer for one Area: it renders the Editor's Toolbars,
the Editor's content widget, tabs (when there are multiple Editors),
and a placeholder (when the Area is empty).

Responsibilities:

- Reflect the Area model. The Area is the source of truth. The
  AreaWidget never modifies the Area.
- Emit signals when the user requests a change (close an Editor,
  add an Editor, change the active Editor). The signals have no
  consumers in this piece; they will be connected by the Workspace
  in a future piece.
- Consume ToolRegistry and ThemeService to materialize Toolbars.

Non-responsibilities:

- The AreaWidget does not create Editors. The Editors come from the
  LayoutBuilder (Piece 4), already initialized.
- The AreaWidget does not mutate the Area. All mutations are
  performed by higher-level components in response to signals.
- The AreaWidget does not know about the LayoutTree, the
  LayoutRenderer, or the Workspace.

Structure of the produced widget:

- Empty Area:    a centered QLabel with "{title} (empty)".
- 1 Editor:      a QVBoxLayout with the Editor's Toolbars and the
                 Editor's content_widget().
- >= 2 Editors:  a QTabWidget (tabs at the bottom) with one tab per
                 Editor. Each tab contains the Editor's Toolbars and
                 its content_widget(). Tabs are closable. A '+' corner
                 button is present for future use.
"""

from __future__ import annotations

from PySide6.QtCore import Qt, Signal
from PySide6.QtGui import QColor, QPalette
from PySide6.QtWidgets import (
    QLabel,
    QTabWidget,
    QVBoxLayout,
    QWidget,
)

from cranioz.ui.editors.base_editor import Editor
from cranioz.ui.theme.service import ThemeService
from cranioz.ui.toolbars.base import Toolbar
from cranioz.ui.tools.registry.tool_definition import ToolContext
from cranioz.ui.tools.registry.tool_registry import ToolRegistry
from cranioz.ui.workspace.areas.area import Area
from cranioz.ui.workspace.areas.placement import Placement


class AreaWidget(QWidget):
    """Qt widget that materializes a single Area.

    The widget reflects the Area model. It never modifies the Area.
    User actions that would change the Area are emitted as signals.
    """

    active_editor_changed = Signal(str, str)  # area_id, editor_id
    editor_close_requested = Signal(str, str)  # area_id, editor_id
    editor_add_requested = Signal(str)  # area_id

    def __init__(
        self,
        area: Area,
        tool_registry: ToolRegistry,
        theme_service: ThemeService,
        debug_colors: bool = False,
        parent: QWidget | None = None,
    ) -> None:
        super().__init__(parent)
        self._area = area
        self._tool_registry = tool_registry
        self._theme_service = theme_service

        self.setObjectName(f"area_widget.{area.area_id}")

        # The root layout is created once and reused. refresh()
        # clears its children and rebuilds them.
        self._root_layout = QVBoxLayout(self)
        self._root_layout.setContentsMargins(0, 0, 0, 0)
        self._root_layout.setSpacing(0)

        if debug_colors:
            self._apply_debug_background()

        self.refresh()

    # ----- Public API -----------------------------------------------

    @property
    def area_id(self) -> str:
        """The area_id of the Area this widget reflects."""
        return self._area.area_id

    def area_placement(self) -> Placement:
        """Return the Placement of the Area this widget reflects.

        Exposed so that the LayoutRenderer can group AreaWidgets by
        Placement when computing Region Bounds.
        """
        return self._area.placement

    def refresh(self) -> None:
        """Re-render the widget from the Area model.

        Called when the Area's Editors or active Editor change.
        The method is idempotent: calling it twice with the same
        Area produces the same widget structure.
        """
        self._clear_layout()

        if self._area.is_empty():
            self._root_layout.addWidget(self._build_empty_placeholder())
            return

        editors = self._area.editors()
        if len(editors) == 1:
            self._root_layout.addWidget(self._build_editor_content(editors[0]))
            return

        self._root_layout.addWidget(self._build_tabs(editors))

    # ----- Internals ------------------------------------------------

    def _apply_debug_background(self) -> None:
        """Apply a soft background color based on Placement.

        Used only for visual debugging. The color makes the
        boundary of each Area visible so that layout issues can be
        spotted at a glance. It does not affect production builds
        when ``debug_colors`` is False.
        """
        palette = {
            Placement.LEFT: "#dbe9f4",  # soft blue
            Placement.CENTRAL: "#dff0d8",  # soft green
            Placement.RIGHT: "#fcf3cf",  # soft yellow
            Placement.BOTTOM: "#e8daef",  # soft lavender
        }
        color = palette.get(self._area.placement)
        if color is None:
            return

        # 1. Palette so the widget paints the color even under dark
        #    OS themes.
        qcolor = QColor(color)
        pal = self.palette()
        pal.setColor(QPalette.ColorRole.Window, qcolor)
        self.setPalette(pal)

        # 2. Auto-fill so Qt paints the Window role.
        self.setAutoFillBackground(True)

        # 3. Styled background + stylesheet, so the color survives
        #    inside a QSplitter.
        self.setAttribute(Qt.WidgetAttribute.WA_StyledBackground, True)
        self.setStyleSheet(f"QWidget#{self.objectName()} {{ background-color: {color}; }}")

    def _clear_layout(self) -> None:
        """Remove all widgets from the root layout and dispose them.

        Uses deleteLater() so that Qt can clean up after the current
        event loop iteration. This is safe even during refresh(),
        which may be triggered from a signal handler.
        """
        while self._root_layout.count():
            item = self._root_layout.takeAt(0)
            widget = item.widget() if item is not None else None
            if widget is not None:
                widget.setParent(None)
                widget.deleteLater()

    def _build_empty_placeholder(self) -> QWidget:
        """Build the placeholder for an empty Area.

        A centered QLabel with "{title} (empty)". The placeholder is
        intentionally minimal for this piece; a '+' button to add an
        Editor will be added in a future piece.
        """
        label = QLabel(f"{self._area.title} (empty)")
        label.setObjectName(f"area_widget.{self._area.area_id}.placeholder")
        label.setAlignment(Qt.AlignmentFlag.AlignCenter)
        return label

    def _build_editor_content(self, editor: Editor) -> QWidget:
        """Build the content container for a single Editor.

        The container holds the Editor's Toolbars (in declared order)
        followed by the Editor's content_widget().
        """
        container = QWidget()
        container.setObjectName(f"area_widget.{self._area.area_id}.editor.{editor.editor_id}")

        layout = QVBoxLayout(container)
        layout.setContentsMargins(0, 0, 0, 0)
        layout.setSpacing(0)

        for toolbar_widget in self._build_toolbars(editor):
            layout.addWidget(toolbar_widget)

        layout.addWidget(editor.content_widget(), stretch=1)
        return container

    def _build_toolbars(self, editor: Editor) -> list[Toolbar]:
        """Materialize the Editor's Toolbars from its ToolbarSpecs.

        Each ToolbarSpec is resolved against the ToolRegistry and the
        ThemeService. The ToolContext is taken from the Editor: the
        Editor is the owner of the ToolContext, not the AreaWidget.
        """
        context = self._editor_tool_context(editor)
        result: list[Toolbar] = []
        for spec in editor.toolbar_specs():
            toolbar = Toolbar(
                spec=spec,
                registry=self._tool_registry,
                theme=self._theme_service,
                context=context,
            )
            result.append(toolbar)
        return result

    @staticmethod
    def _editor_tool_context(editor: Editor) -> ToolContext:
        """Return the ToolContext for an Editor.

        For now, this is an empty context. When Editors expose their
        own ToolContext (via EditorContext or a dedicated accessor),
        this method will read from the Editor instead of constructing
        an empty one.
        """
        return ToolContext()

    def _build_tabs(self, editors: tuple[Editor, ...]) -> QTabWidget:
        """Build the tabbed container for multiple Editors.

        Tabs are placed at the bottom (South). Each tab contains the
        Editor's Toolbars and its content_widget(). Tabs are closable.
        The active tab corresponds to the Area's active_editor_id.
        """
        tabs = QTabWidget()
        tabs.setObjectName(f"area_widget.{self._area.area_id}.tabs")
        tabs.setTabPosition(QTabWidget.TabPosition.South)
        tabs.setTabsClosable(True)
        tabs.setDocumentMode(True)

        # Map editor_id -> index for restoring the active tab.
        index_by_editor_id: dict[str, int] = {}

        for editor in editors:
            content = self._build_editor_content(editor)
            index = tabs.addTab(content, editor.title)
            index_by_editor_id[editor.editor_id] = index

        tabs.tabCloseRequested.connect(self._on_tab_close_requested)
        tabs.currentChanged.connect(self._on_current_tab_changed)

        active_editor_id = self._area.active_editor_id
        if active_editor_id is not None and active_editor_id in index_by_editor_id:
            tabs.setCurrentIndex(index_by_editor_id[active_editor_id])

        return tabs

    # ----- Signal handlers ------------------------------------------

    def _on_tab_close_requested(self, index: int) -> None:
        """Forward a tab close request as a signal.

        The AreaWidget does not remove the Editor. It only emits the
        intent. A higher-level component (the Workspace) will decide
        what to do, likely by dispatching a Command.
        """
        tabs = self._current_tabs()
        if tabs is None:
            return
        editor_id = self._editor_id_for_tab_index(tabs, index)
        if editor_id is None:
            return
        self.editor_close_requested.emit(self._area.area_id, editor_id)

    def _on_current_tab_changed(self, index: int) -> None:
        """Forward a tab change as a signal.

        Emits (area_id, editor_id) so that a higher-level component
        can record the new active Editor. The AreaWidget does not
        mutate the Area.
        """
        tabs = self._current_tabs()
        if tabs is None:
            return
        editor_id = self._editor_id_for_tab_index(tabs, index)
        if editor_id is None:
            return
        self.active_editor_changed.emit(self._area.area_id, editor_id)

    # ----- Helpers --------------------------------------------------

    def _current_tabs(self) -> QTabWidget | None:
        """Return the QTabWidget currently hosted by this widget, if any."""
        for i in range(self._root_layout.count()):
            item = self._root_layout.itemAt(i)
            widget = item.widget() if item is not None else None
            if isinstance(widget, QTabWidget):
                return widget
        return None

    def _editor_id_for_tab_index(
        self,
        tabs: QTabWidget,
        index: int,
    ) -> str | None:
        """Return the editor_id associated with a given tab index.

        The mapping is done by consulting the Area: the Editors are in
        the same order as the tabs, since both derive from
        Area.editors().
        """
        editors = self._area.editors()
        if index < 0 or index >= len(editors):
            return None
        return editors[index].editor_id

"""LayoutBuilder.

Materializes a ModuleUISpec into pure domain structures:

- Instantiates Editors via the EditorRegistry.
- Creates Areas from the AreaSpecs.
- Builds the LayoutTree via LayoutTree.from_specs.
- Registers the Areas in an AreaManager.

The LayoutBuilder does NOT touch Qt. It does not create widgets, does
not create Toolbars, does not materialize anything visual. That
responsibility belongs to the LayoutRenderer (see ADR-0002).

Contract (see ADR-0002):

- For each AreaSpec, the LayoutBuilder walks its EditorLayoutSpec in
  depth-first order and creates one Editor instance per editor_id.
- Each Editor receives its own EditorContext. The editor_id is specific
  to that Editor; extras are copied per Editor, so mutations do not
  leak between Editors (hybrid EditorContext model, ADR-0002 §2.1).
- The EditorLayoutSpec is validated structurally by its own dataclass
  at construction time. The LayoutBuilder does not repeat that
  validation.
- An unknown editor_id raises KeyError, propagated from the
  EditorRegistry with a clear message.
- The LayoutBuilder never imports PySide6 and never requires a
  QApplication (ADR-0002 §2.2).

Extension point (ADR-0002 §2.3):

- build() accepts an optional ``shared_context`` mapping. Its values
  are intended to be propagated by reference into every EditorContext
  created during the build.

  Today, the EditorContext dataclass has no fields to receive shared
  services (SceneQuery, CommandDispatcher, EventSubscription). The
  shared_context parameter is accepted and validated for type, but
  its contents are not yet propagated. When the first shared service
  is implemented, EditorContext will gain the corresponding typed
  fields and this builder will propagate them by reference. The
  parameter exists now to keep the builder interface stable across
  that evolution.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any

from cranioz.ui.editors.base_editor import Editor
from cranioz.ui.editors.editor_context import EditorContext
from cranioz.ui.editors.editor_registry import EditorRegistry
from cranioz.ui.workspace.areas.area import Area
from cranioz.ui.workspace.areas.area_manager import AreaManager
from cranioz.ui.workspace.layout.layout_tree import LayoutTree
from cranioz.ui.workspace.specs.area_spec import AreaSpec
from cranioz.ui.workspace.specs.editor_layout_spec import (
    EditorLayoutSpec,
    LayoutType,
)
from cranioz.ui.workspace.specs.module_ui_spec import ModuleUISpec


class LayoutBuilderError(ValueError):
    """Raised when the LayoutBuilder cannot materialize a spec."""


@dataclass(frozen=True)
class BuildResult:
    """Result of materializing a ModuleUISpec.

    Contains pure domain structures:

    - The Areas that were created.
    - The LayoutTree that was built.
    - The AreaManager, already populated with the Areas.
    - The Editors, indexed by the area_id of the Area that hosts them.

    Because the same editor_id may appear in multiple Areas (as two
    independent instances), a single flat list of Editors would lose
    the association between Editor and Area. The ``editors_by_area``
    mapping preserves that association.
    """

    areas: tuple[Area, ...]
    tree: LayoutTree
    area_manager: AreaManager
    editors_by_area: dict[str, tuple[Editor, ...]] = field(default_factory=dict)

    def area(self, area_id: str) -> Area:
        """Return the Area with the given area_id.

        Raises KeyError if no such Area exists.
        """
        return self.area_manager.require(area_id)

    def editors(self, area_id: str) -> tuple[Editor, ...]:
        """Return the Editors hosted by the Area with the given area_id.

        Returns an empty tuple if the Area exists but hosts no Editors.
        Raises KeyError if no such Area exists.
        """
        if not self.area_manager.contains(area_id):
            raise KeyError(f"area_id not registered: {area_id}")
        return self.editors_by_area.get(area_id, ())


class LayoutBuilder:
    """Builds pure domain structures from a ModuleUISpec.

    The builder is stateless with respect to a single build(): it can
    be reused across builds. State that varies per build (for example,
    the context extras for the Editors) is passed as arguments to
    build(), not stored on the builder.
    """

    def __init__(self, editor_registry: EditorRegistry) -> None:
        self._registry = editor_registry

    def build(
        self,
        spec: ModuleUISpec,
        *,
        context_extras: dict[str, Any] | None = None,
        shared_context: dict[str, Any] | None = None,
    ) -> BuildResult:
        """Materialize the ModuleUISpec.

        Args:
            spec: the declarative UI composition.
            context_extras: extras propagated into each EditorContext.
                Each Editor receives a copy, so mutations do not leak
                between Editors (ADR-0002 §2.1).
            shared_context: extension point reserved for shared
                services (SceneQuery, CommandDispatcher,
                EventSubscription). Accepted and validated for type,
                but not yet propagated, because EditorContext has no
                fields to receive them (ADR-0002 §2.3). Passing a
                non-mapping raises TypeError.

        Returns:
            A BuildResult containing the Areas, the LayoutTree, the
            populated AreaManager, and the Editors indexed by Area.

        Raises:
            LayoutBuilderError: if the spec cannot be materialized.
            KeyError: if an editor_id is not registered in the
                EditorRegistry.
            TypeError: if context_extras or shared_context is not a
                mapping.
        """
        if context_extras is not None and not isinstance(context_extras, dict):
            raise TypeError(
                f"context_extras must be a dict or None, got {type(context_extras).__name__}"
            )
        if shared_context is not None and not isinstance(shared_context, dict):
            raise TypeError(
                f"shared_context must be a dict or None, got {type(shared_context).__name__}"
            )

        # Defensive copies: the builder does not retain references to
        # caller-owned mutable state beyond the build() call.
        extras = dict(context_extras) if context_extras else {}
        # shared_context is intentionally not copied. When it becomes
        # functional (ADR-0002 §2.3), it will be propagated by
        # reference to preserve resource identity across Editors.
        # Today it is accepted but not used; we discard it explicitly
        # to avoid an unused-argument warning and to keep intent clear.
        _ = shared_context

        # 1. Instantiate Editors and create Areas.
        areas: list[Area] = []
        areas_by_id: dict[str, Area] = {}
        editors_by_area: dict[str, tuple[Editor, ...]] = {}

        for area_spec in spec.areas:
            editors = self._instantiate_editors(area_spec, extras)

            area = Area(
                area_id=area_spec.area_id,
                title=area_spec.title,
                placement=area_spec.placement,
            )
            for editor in editors:
                area.add_editor(editor)

            areas.append(area)
            areas_by_id[area.area_id] = area
            editors_by_area[area.area_id] = tuple(editors)

        # 2. Build the LayoutTree.
        #    from_specs validates that at least one Area is CENTRAL and
        #    that every AreaSpec references a known area_id. It also
        #    preserves the grouping of LEFT, CENTRAL, RIGHT, BOTTOM.
        try:
            tree = LayoutTree.from_specs(tuple(spec.areas), areas_by_id)
        except ValueError as exc:
            # Re-raise as LayoutBuilderError so callers of the builder
            # catch a single error type, without losing the cause.
            raise LayoutBuilderError(str(exc)) from exc

        # 3. Register the Areas in the AreaManager.
        manager = AreaManager()
        for area in areas:
            manager.register(area)

        return BuildResult(
            areas=tuple(areas),
            tree=tree,
            area_manager=manager,
            editors_by_area=editors_by_area,
        )

    # ----- Internals ------------------------------------------------

    def _instantiate_editors(
        self,
        area_spec: AreaSpec,
        extras: dict[str, Any],
    ) -> list[Editor]:
        """Walk the EditorLayoutSpec depth-first and create one Editor
        per LEAF, each with its own EditorContext.

        The EditorContext is per Editor (ADR-0002 §2.1). The extras are
        copied per Editor so that mutations do not leak between them.
        """
        editor_ids = self._collect_editor_ids(area_spec.editor_layout)
        editors: list[Editor] = []

        for editor_id in editor_ids:
            # The EditorRegistry resolves the id to a class and creates
            # an instance. KeyError propagates with a clear message.
            editor = self._registry.create(
                editor_id,
                editor_id=editor_id,
                title=editor_id,
            )

            # A fresh EditorContext per Editor. extras is copied so
            # that two Editors with the same editor_id do not share
            # mutable state.
            context = EditorContext(
                editor_id=editor_id,
                extras=dict(extras),
            )
            editor.initialize(context)
            editors.append(editor)

        return editors

    @staticmethod
    def _collect_editor_ids(layout: EditorLayoutSpec) -> list[str]:
        """Collect editor_ids in depth-first order from an
        EditorLayoutSpec.

        The EditorLayoutSpec dataclass guarantees:
        - A LEAF node has a non-empty editor_id and no children.
        - A SPLIT node has a direction and at least two children.
        This method relies on those invariants and does not repeat them.
        """
        result: list[str] = []

        def visit(node: EditorLayoutSpec) -> None:
            if node.type is LayoutType.LEAF:
                editor_id = node.editor_id
                if editor_id is None:
                    raise LayoutBuilderError(
                        "LEAF EditorLayoutSpec has no editor_id; "
                        "this violates the EditorLayoutSpec invariant"
                    )
                result.append(editor_id)
                return
            for child in node.children:
                visit(child)

        visit(layout)
        return result

"""LayoutRenderer.

Materializes a LayoutTree into Qt widgets.

The LayoutRenderer is a translator between the pure data model
(LayoutTree, produced by the LayoutBuilder) and Qt. It never mutates
the LayoutTree.

The renderer maintains a rebuildable cache of widgets:

- split_widgets: id(Split)  -> QSplitter
- area_widgets:  area_id    -> AreaWidget

The cache is an optimization. It is always rebuildable from the
tree. If there is any divergence between the cache and the tree,
the tree wins. The cache exists so that incremental re-renders (for
example, when a single Area changes) can reuse existing widgets
instead of rebuilding everything.

Responsibilities:

- Walk the LayoutTree recursively.
- For each Leaf, obtain or create an AreaWidget for the Area.
- For each Split, obtain or create a QSplitter with the correct
  orientation, and insert the rendered children.
- Return the root widget.

Non-responsibilities:

- The LayoutRenderer does not create Editors. Editors come from the
  LayoutBuilder, already initialized.
- The LayoutRenderer does not materialize the interior of an Area.
  That is the AreaWidget's job.
- The LayoutRenderer does not manage overlays or regions. Those are
  handled in later pieces.
- The LayoutRenderer does not persist layout state. That is
  handled in a later piece.

Qt dependency:

- The LayoutRenderer imports PySide6. It is the first piece of the
  Workspace layout subsystem that requires a QApplication to run.
- Its tests use qtbot and instantiate real widgets.
"""

from __future__ import annotations

from PySide6.QtCore import Qt
from PySide6.QtWidgets import QSplitter, QWidget

from cranioz.ui.theme.service import ThemeService
from cranioz.ui.tools.registry.tool_registry import ToolRegistry
from cranioz.ui.workspace.areas.area_widget import AreaWidget
from cranioz.ui.workspace.areas.placement import Placement
from cranioz.ui.workspace.layout.layout_tree import (
    LayoutNode,
    LayoutTree,
    Leaf,
    Split,
)
from cranioz.ui.workspace.overlays.region import Bounds
from cranioz.ui.workspace.specs.editor_layout_spec import SplitDirection


class LayoutRendererError(ValueError):
    """Raised when the LayoutRenderer cannot materialize a tree."""


class LayoutRenderer:
    """Materializes a LayoutTree into Qt widgets.

    The renderer is stateless with respect to a single render() call:
    it can be reused across renders. State that varies between renders
    (the widget cache) is stored on the instance but is always
    rebuildable from the tree.
    """

    def __init__(
        self,
        tool_registry: ToolRegistry,
        theme_service: ThemeService,
        debug_colors: bool = False,
    ) -> None:
        self._tool_registry = tool_registry
        self._theme_service = theme_service
        self._debug_colors = debug_colors

        # Caches. Keyed by the identity of the model object (id(Split))
        # or by the stable domain identifier (area_id).
        self._split_widgets: dict[int, QSplitter] = {}
        self._area_widgets: dict[str, AreaWidget] = {}

    # ----- Public API -----------------------------------------------

    def render(self, tree: LayoutTree) -> QWidget:
        """Render the LayoutTree and return the root widget.

        The root widget is either:

        - a QSplitter, when the root node is a Split;
        - an AreaWidget, when the root node is a Leaf.

        The method is idempotent for the same tree instance: calling
        it twice returns the same root widget, reusing cached
        sub-widgets.
        """
        return self._render_node(tree.root)

    def clear_cache(self) -> None:
        """Discard all cached widgets.

        Called when the tree is fully rebuilt and old widgets are no
        longer needed. After clear_cache(), a subsequent render()
        creates new widgets from scratch.
        """
        for splitter in self._split_widgets.values():
            splitter.setParent(None)
            splitter.deleteLater()
        for area_widget in self._area_widgets.values():
            area_widget.setParent(None)
            area_widget.deleteLater()
        self._split_widgets.clear()
        self._area_widgets.clear()

    def cached_area_widget(self, area_id: str) -> AreaWidget | None:
        """Return the cached AreaWidget for the given area_id, if any."""
        return self._area_widgets.get(area_id)

    # ----- Region geometry ------------------------------------------

    def compute_region_bounds(
        self,
    ) -> dict[Placement, Bounds]:
        """Compute the geometric Bounds of each Region.

        A Region groups every Area that shares the same Placement.
        Its Bounds is the axis-aligned union of the geometries of
        the AreaWidgets that belong to it.

        The method reads the current geometry of the cached
        AreaWidgets. It therefore produces meaningful results only
        after ``render()`` has been called and Qt has assigned the
        widgets their real geometry (typically after the parent
        widget has been shown or laid out).

        Placements with no AreaWidget in the cache produce no
        entry in the result. Callers that need a stable shape may
        use ``regions_from_bounds()`` (see below) to build a list
        of Regions in a documented order.
        """
        # Group the AreaWidgets by their Area's Placement.
        grouped: dict[Placement, list[AreaWidget]] = {}
        for widget in self._area_widgets.values():
            # The AreaWidget exposes its area_id; we need the
            # Placement. We look it up through the widget's Area,
            # which is stored inside the widget.
            placement = widget.area_placement()
            grouped.setdefault(placement, []).append(widget)

        # Compute the union rectangle for each Placement.
        result: dict[Placement, Bounds] = {}
        for placement, widgets in grouped.items():
            union = self._union_bounds(widgets)
            if union is not None:
                result[placement] = union
        return result

    @staticmethod
    def _union_bounds(widgets: list[AreaWidget]) -> Bounds | None:
        """Return the axis-aligned union of the widgets' geometries.

        Returns None when the list is empty. Coordinates are in the
        local coordinate space of each widget's parent; the union
        assumes the widgets share a common parent, which is the
        case for all AreaWidgets inside the same Workspace.
        """
        if not widgets:
            return None

        rects = [widget.geometry() for widget in widgets]
        left = min(r.left() for r in rects)
        top = min(r.top() for r in rects)
        right = max(r.right() for r in rects)
        bottom = max(r.bottom() for r in rects)

        return Bounds(
            x=float(left),
            y=float(top),
            width=float(right - left),
            height=float(bottom - top),
        )

    # ----- Internals ------------------------------------------------

    def _render_node(self, node: LayoutNode) -> QWidget:
        """Dispatch a node to the leaf or split renderer."""
        if isinstance(node, Leaf):
            return self._render_leaf(node)
        if isinstance(node, Split):
            return self._render_split(node)
        raise LayoutRendererError(f"unknown layout node type: {type(node).__name__}")

    def _render_leaf(self, leaf: Leaf) -> AreaWidget:
        """Return (or create) an AreaWidget for the leaf's Area."""
        area = leaf.area
        widget = self._area_widgets.get(area.area_id)
        if widget is None:
            widget = AreaWidget(
                area=area,
                tool_registry=self._tool_registry,
                theme_service=self._theme_service,
                debug_colors=self._debug_colors,
            )
            self._area_widgets[area.area_id] = widget
        return widget

    def _render_split(self, split: Split) -> QSplitter:
        """Return (or create) a QSplitter for the split node.

        The QSplitter is created once per Split node (keyed by
        id(split)) and reused on subsequent renders of the same tree.
        Children are always re-inserted in the correct order, so the
        splitter reflects the current structure of the tree.
        """
        key = id(split)
        splitter = self._split_widgets.get(key)
        if splitter is None:
            splitter = QSplitter(self._orientation_for(split.direction))
            splitter.setObjectName(f"split_{key}")
            splitter.setChildrenCollapsible(False)
            self._split_widgets[key] = splitter

        # Ensure children are present and in the right order. We do
        # not recreate children that are already attached to the
        # splitter, to preserve their internal state (e.g. tab
        # selection inside an AreaWidget).
        self._reconcile_children(splitter, split)
        return splitter

    def _reconcile_children(
        self,
        splitter: QSplitter,
        split: Split,
    ) -> None:
        """Ensure the splitter's children match the split's children.

        Rules:
        - A child widget that is already attached to the splitter is
          left in place (its internal state is preserved).
        - A child widget that is missing is created and inserted in
          the right position.
        - A child widget that is attached but no longer part of the
          split is detached (its parent is cleared, but the widget is
          not destroyed — it may be reused elsewhere, or it lives in
          the area_widgets cache).
        """
        expected_widgets: list[QWidget] = [self._render_node(child) for child in split.children]

        # Collect the widgets currently attached to the splitter.
        # QSplitter.widget(i) is typed as QWidget | None, so we filter
        # out None values explicitly. This is not merely a typing fix:
        # it protects the loop below from a None that would otherwise
        # break .setParent().
        current_widgets: list[QWidget] = []
        for i in range(splitter.count()):
            widget = splitter.widget(i)
            if widget is not None:
                current_widgets.append(widget)

        # Detach widgets that should no longer be children of this
        # splitter. We do not destroy them here: they may still be in
        # the area_widgets cache and reused later.
        for widget in current_widgets:
            if widget not in expected_widgets:
                widget.setParent(None)

        # Insert the expected widgets in the right order. Qt's
        # QSplitter inserts at the requested index when the widget is
        # not yet a child. If the widget is already a child at the
        # correct index, insertWidget() is a no-op for the position.
        for index, widget in enumerate(expected_widgets):
            current_index = splitter.indexOf(widget)
            if current_index == index:
                continue
            splitter.insertWidget(index, widget)

    @staticmethod
    def _orientation_for(direction: SplitDirection) -> Qt.Orientation:
        """Map a SplitDirection to a Qt orientation.

        HORIZONTAL in CranioZ means "children are laid out
        side by side, from left to right". That corresponds to
        Qt.Horizontal (QSplitter splits horizontally into columns).

        VERTICAL means "children are stacked top to bottom". That
        corresponds to Qt.Vertical.
        """
        if direction is SplitDirection.HORIZONTAL:
            return Qt.Orientation.Horizontal
        if direction is SplitDirection.VERTICAL:
            return Qt.Orientation.Vertical
        raise LayoutRendererError(f"unknown split direction: {direction!r}")

"""LayoutTree.

N-ary tree that organizes Areas within a Workspace.

Node types:
- Leaf: contains exactly one Area.
- Split: contains two or more children (Leaf or Split) and a direction
  (HORIZONTAL or VERTICAL).

The LayoutTree is the single source of truth for structural layout.
It does not carry proportions, visibility, or active-Editor state;
those live in LayoutState.

Rules:
- A Leaf must contain an Area.
- A Split must have a direction and at least two children.
- A LayoutTree must have a root node.
- from_specs() requires at least one CENTRAL AreaSpec.
- from_specs() preserves the order of AreaSpecs in the resulting
  tree, grouped by placement:
    left-most areas → central areas → right-most areas, then bottom.
  The exact grouping algorithm is defined by the builder; the tree
  itself does not interpret placement.
"""

from __future__ import annotations

from dataclasses import dataclass

from cranioz.ui.workspace.areas.area import Area
from cranioz.ui.workspace.areas.placement import Placement
from cranioz.ui.workspace.overlays.region import Bounds, Region
from cranioz.ui.workspace.specs.area_spec import AreaSpec
from cranioz.ui.workspace.specs.editor_layout_spec import SplitDirection


class LayoutTreeError(ValueError):
    """Raised when a LayoutTree violates a structural rule."""


# ----- Nodes -------------------------------------------------------


@dataclass(frozen=True)
class Leaf:
    """A LayoutTree node that contains exactly one Area."""

    area: Area


@dataclass(frozen=True)
class Split:
    """A LayoutTree node that contains two or more children."""

    direction: SplitDirection
    children: tuple[Leaf | Split, ...]

    def __post_init__(self) -> None:
        if len(self.children) < 2:
            raise LayoutTreeError("Split must have at least two children")


LayoutNode = Leaf | Split


# ----- Tree --------------------------------------------------------


class LayoutTree:
    """The N-ary split tree that organizes Areas in a Workspace."""

    def __init__(self, root: LayoutNode) -> None:
        self._root = root

    # ----- Access ---------------------------------------------------

    @property
    def root(self) -> LayoutNode:
        return self._root

    def leaves(self) -> tuple[Leaf, ...]:
        """Return all Leaves in depth-first order."""
        result: list[Leaf] = []
        self._collect_leaves(self._root, result)
        return tuple(result)

    def areas(self) -> tuple[Area, ...]:
        """Return all Areas in depth-first order."""
        return tuple(leaf.area for leaf in self.leaves())

    def find(self, area_id: str) -> Leaf | None:
        """Return the Leaf that contains the given area_id, or None."""
        for leaf in self.leaves():
            if leaf.area.area_id == area_id:
                return leaf
        return None

    def contains(self, area_id: str) -> bool:
        """Return True if the tree contains an Area with the given id."""
        return self.find(area_id) is not None

    def area_count(self) -> int:
        """Return the number of Areas in the tree."""
        return len(self.leaves())

    # ----- Construction from specs ----------------------------------

    @classmethod
    def from_specs(
        cls,
        specs: tuple[AreaSpec, ...],
        areas: dict[str, Area],
    ) -> LayoutTree:
        """Build a LayoutTree from a list of AreaSpecs and a map of Areas.

        The map `areas` provides the concrete Area instances keyed by
        area_id. The caller (LayoutBuilder) is responsible for creating
        them from the specs.

        Rules:
        - At least one spec must have Placement.CENTRAL.
        - Every spec.area_id must exist in the `areas` map.

        Layout grouping:
        - LEFT specs (in order) → CENTRAL specs (in order) → RIGHT specs
          (in order) form a HORIZONTAL row, if any side area exists.
        - If BOTTOM specs exist, the tree becomes a VERTICAL split of
          the row and the bottom row.
        - If there is only one Area total, the tree is a single Leaf.
        """
        if not specs:
            raise LayoutTreeError("LayoutTree.from_specs requires at least one spec")

        for spec in specs:
            if spec.area_id not in areas:
                raise LayoutTreeError(f"AreaSpec references an unknown area_id: {spec.area_id}")

        left = [s for s in specs if s.placement is Placement.LEFT]
        central = [s for s in specs if s.placement is Placement.CENTRAL]
        right = [s for s in specs if s.placement is Placement.RIGHT]
        bottom = [s for s in specs if s.placement is Placement.BOTTOM]

        if not central:
            raise LayoutTreeError("LayoutTree.from_specs requires at least one CENTRAL AreaSpec")

        # Top row: LEFT + CENTRAL + RIGHT, in that order.
        top_row_specs = [*left, *central, *right]

        top_node: LayoutNode
        if len(top_row_specs) == 1:
            top_node = Leaf(area=areas[top_row_specs[0].area_id])
        else:
            top_node = Split(
                direction=SplitDirection.HORIZONTAL,
                children=tuple(Leaf(area=areas[s.area_id]) for s in top_row_specs),
            )

        if not bottom:
            return cls(top_node)

        bottom_row_specs = bottom
        bottom_node: LayoutNode
        if len(bottom_row_specs) == 1:
            bottom_node = Leaf(area=areas[bottom_row_specs[0].area_id])
        else:
            bottom_node = Split(
                direction=SplitDirection.HORIZONTAL,
                children=tuple(Leaf(area=areas[s.area_id]) for s in bottom_row_specs),
            )

        root = Split(
            direction=SplitDirection.VERTICAL,
            children=(top_node, bottom_node),
        )
        return cls(root)

    # ----- Internals ------------------------------------------------

    @staticmethod
    def _collect_leaves(node: LayoutNode, out: list[Leaf]) -> None:
        if isinstance(node, Leaf):
            out.append(node)
            return
        for child in node.children:
            LayoutTree._collect_leaves(child, out)

    # ----- Representation ------------------------------------------

    def __repr__(self) -> str:
        return f"LayoutTree(root={self._root!r})"

    # ----- Regions --------------------------------------------------

    def find_regions(self) -> tuple[Region, ...]:
        """Group the Areas of this tree into Regions by Placement.

        The method is purely structural: it does not compute any
        geometric bounds. Every returned Region has a placeholder
        Bounds (zero rectangle). The real geometry is filled in by
        the rendering layer once the widgets have been materialized.

        Regions are returned in a stable, documented order:

            LEFT, CENTRAL, RIGHT, BOTTOM

        A Placement that has no Area in this tree produces no Region.
        Every Area in the tree belongs to exactly one Region.
        """
        regions: list[Region] = []
        for placement in (
            Placement.LEFT,
            Placement.CENTRAL,
            Placement.RIGHT,
            Placement.BOTTOM,
        ):
            areas = tuple(leaf.area for leaf in self.leaves() if leaf.area.placement is placement)
            if not areas:
                continue
            regions.append(
                Region(
                    region_id=placement.value,
                    bounds=Bounds(),
                    areas=areas,
                    semantic_role=placement,
                )
            )
        return tuple(regions)

"""Overlay.

An Overlay is a floating layer managed by the Workspace. It is
positioned over a Region, not over an individual Area. The Overlay
does not participate in the layout tree; it floats above the Areas.

The distinction matters (see workspace_pt.md §7.1):

- An Area is part of the layout tree; it is arranged by the
  LayoutRenderer into QSplitters.
- An Overlay is a separate widget that the Workspace places over a
  Region. It is not hosted by any Area. It can be shown, hidden,
  moved, and resized without touching the layout tree.

The Overlay ABC defines the minimal contract the Workspace needs to
manage a floating layer:

- A stable identity (``overlay_id``).
- A target Region, identified by its ``Placement``.
- Access to the Qt widget that represents the Overlay.
- Ability to reposition the widget when the Region's Bounds change.
- Ability to show / hide the widget.

Concrete Overlays (e.g. the Top Bar, the Steps Panel) live in their
own modules and implement this contract. The Workspace only knows
the ABC.

Qt dependency:

- ``Overlay.widget()`` returns a ``QWidget``, so concrete Overlays
  must import PySide6. That is expected: an Overlay is a visual
  layer. The ABC itself imports ``QWidget`` only for typing; it does
  not instantiate Qt objects.
- The ABC is still safe to import without a QApplication, as long
  as no concrete Overlay is instantiated. Tests that instantiate a
  concrete Overlay use ``qtbot`` and a running QApplication.
"""

from __future__ import annotations

from abc import ABC, abstractmethod
from typing import TYPE_CHECKING

from cranioz.ui.workspace.areas.placement import Placement
from cranioz.ui.workspace.overlays.region import Bounds

if TYPE_CHECKING:
    from PySide6.QtWidgets import QWidget


class OverlayError(ValueError):
    """Raised when an Overlay operation violates a structural rule."""


class Overlay(ABC):
    """Abstract base class for a floating layer managed by the Workspace.

    An Overlay is identified by a stable ``overlay_id`` and targets
    a Region identified by its ``Placement``. The Workspace uses the
    target to decide where to place the Overlay as the layout changes.
    """

    # ----- Identity --------------------------------------------------

    @property
    @abstractmethod
    def overlay_id(self) -> str:
        """Return the stable identifier of this Overlay.

        The identifier must be unique within a Workspace. It is
        used by the OverlayManager to register and retrieve the
        Overlay, and by other components to reference it.
        """
        raise NotImplementedError

    @property
    @abstractmethod
    def target_region(self) -> Placement:
        """Return the Placement of the Region this Overlay targets.

        The Workspace positions the Overlay over the Region whose
        ``semantic_role`` matches this Placement. If the Region does
        not exist (e.g. no Area has that Placement), the Overlay is
        not shown.
        """
        raise NotImplementedError

    # ----- Widget ----------------------------------------------------

    @abstractmethod
    def widget(self) -> QWidget:
        """Return the Qt widget that represents this Overlay.

        The Overlay owns the widget. The Workspace does not create,
        destroy, or reparent it. The widget is used as-is by the
        OverlayManager when it positions the Overlay on the screen.

        The method is allowed to return the same instance on every
        call. The Overlay is responsible for managing the widget's
        lifecycle (creation, configuration, disposal).
        """
        raise NotImplementedError

    # ----- Geometry --------------------------------------------------

    @abstractmethod
    def reposition(self, bounds: Bounds) -> None:
        """Reposition the Overlay to the given Bounds.

        The Workspace calls this method whenever the target Region's
        geometry changes (window resize, layout switch, Region
        recomputation). The Overlay applies its own anchoring rules
        within the given Bounds: for example, an Overlay anchored to
        the top of the Region positions itself at ``bounds.y``,
        while an Overlay anchored to the bottom positions itself at
        ``bounds.y + bounds.height - overlay_height``.

        The method must not change the Overlay's identity or target.
        It only adjusts the widget's position and size.
        """
        raise NotImplementedError

    # ----- Visibility ------------------------------------------------

    @abstractmethod
    def set_visible(self, visible: bool) -> None:
        """Show or hide the Overlay.

        Hiding an Overlay does not destroy its widget and does not
        remove it from the OverlayManager. The Overlay remains
        registered; it is simply not shown on screen.
        """
        raise NotImplementedError

    @abstractmethod
    def is_visible(self) -> bool:
        """Return True if the Overlay is currently visible."""
        raise NotImplementedError

"""OverlayManager.

Registry and repositioning coordinator for Overlays within a
Workspace.

Responsibilities:

- Register and unregister Overlays by their ``overlay_id``.
- Look them up and enumerate them in a deterministic order.
- Reposition every registered Overlay when the layout changes, by
  matching each Overlay's ``target_region`` (a ``Placement``) to the
  current set of Regions and passing the Region's Bounds to
  ``Overlay.reposition()``.

Non-responsibilities:

- The manager does not create or destroy Overlays. It only tracks
  the ones it is given.
- The manager does not compute Region Bounds. It receives the
  current Regions from the Workspace / LayoutRenderer and forwards
  the Bounds as-is.
- The manager does not know about Qt layout details. Overlays own
  their widgets.

Design rules:

- A registered ``overlay_id`` is unique within a single manager.
  Registering the same instance again is idempotent; registering a
  different instance under the same id raises
  ``OverlayManagerError``.
- Registration order is preserved. ``overlays()`` returns them in
  that order, which makes repositioning deterministic.
- ``reposition_all()`` does not fail when a target Region is absent.
  If an Overlay targets a Placement for which no Region exists in
  the current tree, the Overlay is simply not repositioned. This
  matches the Workspace model: a Region may legitimately be missing
  during a module switch.
- ``clear()`` removes all registrations. It does not touch the
  Overlays themselves.
"""

from __future__ import annotations

from cranioz.ui.workspace.areas.placement import Placement
from cranioz.ui.workspace.overlays.overlay import Overlay
from cranioz.ui.workspace.overlays.region import Region


class OverlayManagerError(ValueError):
    """Raised when overlay registration violates a structural rule."""


class OverlayManager:
    """Registry and repositioning coordinator for Overlays."""

    def __init__(self) -> None:
        self._overlays: dict[str, Overlay] = {}

    # ----- Registration ---------------------------------------------

    def register(self, overlay: Overlay) -> None:
        """Register an Overlay.

        Idempotent when the same instance is registered twice under
        the same ``overlay_id``. Raises ``OverlayManagerError`` when
        a different instance is already registered under the same id.
        """
        overlay_id = overlay.overlay_id
        if not overlay_id:
            raise OverlayManagerError("overlay_id must not be empty")

        existing = self._overlays.get(overlay_id)
        if existing is overlay:
            return
        if existing is not None:
            raise OverlayManagerError(
                f"overlay_id already registered with a different Overlay: {overlay_id}"
            )
        self._overlays[overlay_id] = overlay

    def unregister(self, overlay_id: str) -> Overlay | None:
        """Remove an Overlay.

        Returns the removed Overlay, or None if the id is unknown.
        """
        return self._overlays.pop(overlay_id, None)

    # ----- Lookup ---------------------------------------------------

    def get(self, overlay_id: str) -> Overlay | None:
        """Return the Overlay with the given id, or None."""
        return self._overlays.get(overlay_id)

    def require(self, overlay_id: str) -> Overlay:
        """Return the Overlay with the given id.

        Raises KeyError if unknown.
        """
        try:
            return self._overlays[overlay_id]
        except KeyError as exc:
            raise KeyError(f"overlay_id not registered: {overlay_id}") from exc

    def contains(self, overlay_id: str) -> bool:
        """Return True if an Overlay with the given id is registered."""
        return overlay_id in self._overlays

    def overlays(self) -> tuple[Overlay, ...]:
        """Return the registered Overlays in registration order."""
        return tuple(self._overlays.values())

    def overlay_count(self) -> int:
        """Return the number of registered Overlays."""
        return len(self._overlays)

    def clear(self) -> None:
        """Remove all registered Overlays.

        The Overlay objects themselves are not destroyed; only the
        manager's references to them are dropped.
        """
        self._overlays.clear()

    # ----- Repositioning --------------------------------------------

    def reposition_all(self, regions: tuple[Region, ...]) -> None:
        """Reposition every registered Overlay according to the Regions.

        For each registered Overlay, this method looks up the Region
        whose ``semantic_role`` matches the Overlay's
        ``target_region``. If such a Region exists, the Overlay is
        repositioned with that Region's Bounds. If no Region matches,
        the Overlay is left untouched.

        The method is a no-op when no Overlays are registered. It
        does not raise if Regions are missing for some Overlays;
        missing targets are a legitimate state during a module
        switch, before the next layout has been computed.

        If two Regions share the same ``semantic_role`` (a
        pathological case that should not occur in practice), the
        first one encountered wins.
        """
        # Build a fast lookup: Placement -> Region.
        # The first Region seen for a given Placement wins, so we
        # explicitly skip assignments that would overwrite an
        # existing entry.
        region_by_placement: dict[Placement, Region] = {}
        for region in regions:
            role = region.semantic_role
            if role is None:
                continue
            if role not in region_by_placement:
                region_by_placement[role] = region

        for overlay in self._overlays.values():
            target = region_by_placement.get(overlay.target_region)
            if target is None:
                continue
            overlay.reposition(target.bounds)

    # ----- Representation -------------------------------------------

    def __repr__(self) -> str:
        return f"OverlayManager(overlays={list(self._overlays.keys())!r})"

"""Region.

A Region is a geometric grouping of Areas within a Workspace.

The concept exists to give Overlays a target: an Overlay is positioned
over a Region, not over an individual Area. This decouples the Overlay
from the current number and arrangement of Areas.

Two concepts are defined here:

- ``Bounds``: a plain geometric rectangle. It is deliberately NOT
  ``PySide6.QtCore.QRect``. The layout tree and the Region are pure
  domain structures; they do not depend on Qt. The LayoutRenderer is
  responsible for converting ``Bounds`` to a Qt geometry when it
  materializes widgets.

- ``Region``: a named group of Areas that share a semantic role
  (LEFT, CENTRAL, RIGHT, BOTTOM).

A Region is *derived* from the LayoutTree, not stored on it. The
LayoutTree owns the Areas; the Region is a projection of them.

Bounds lifecycle:

- ``LayoutTree.find_regions()`` produces Regions with a *placeholder*
  Bounds (zero rectangle). The association between Areas and Regions
  is structural and can be computed without Qt.

- The real Bounds can only be computed after the LayoutRenderer has
  materialized the Area widgets and Qt has assigned them their actual
  geometry. A future piece (OverlayManager or Workspace) will fill
  the Bounds in.

This separation keeps the pure model (LayoutTree, Region) testable
without a QApplication, and confines geometry to the rendering layer.
"""

from __future__ import annotations

from dataclasses import dataclass

from cranioz.ui.workspace.areas.area import Area
from cranioz.ui.workspace.areas.placement import Placement

# ----- Bounds -------------------------------------------------------


@dataclass(frozen=True)
class Bounds:
    """A geometric rectangle in screen coordinates.

    The unit is left unspecified here because Bounds is a pure
    geometry value. In practice it is pixels, as provided by Qt.
    A future piece may introduce a coordinate-system tag if needed.

    A Bounds with all fields at zero is a valid placeholder: it
    means "the real geometry has not been computed yet".
    """

    x: float = 0.0
    y: float = 0.0
    width: float = 0.0
    height: float = 0.0

    def is_placeholder(self) -> bool:
        """Return True if the Bounds has not been populated yet.

        Used by tests and by the OverlayManager to distinguish a
        Region whose geometry is known from one whose geometry has
        not yet been computed.
        """
        return self == _PLACEHOLDER_BOUNDS

    def __post_init__(self) -> None:
        if self.width < 0:
            raise ValueError(f"Bounds.width must be non-negative, got {self.width}")
        if self.height < 0:
            raise ValueError(f"Bounds.height must be non-negative, got {self.height}")


_PLACEHOLDER_BOUNDS = Bounds()


# ----- Region -------------------------------------------------------


@dataclass(frozen=True)
class Region:
    """A named group of Areas that share a semantic role.

    A Region is a *projection* of a subset of a LayoutTree. It is
    produced by ``LayoutTree.find_regions()`` and is intentionally
    immutable: any change to the layout produces a new set of Regions.

    Fields:

    - ``region_id``: stable identifier. It equals the Placement value
      ("left", "central", "right", "bottom"). Using the Placement
      value as the id ensures the Region identity is consistent with
      the semantic role the tree already uses.

    - ``bounds``: geometric rectangle that contains every Area in the
      Region. It is a placeholder until the LayoutRenderer (or the
      Workspace) computes the real geometry.

    - ``areas``: the Areas that belong to the Region, in the order
      they appear in the LayoutTree. The order is deterministic and
      matches the depth-first order of the tree.

    - ``semantic_role``: the Placement this Region was derived from.
      It is guaranteed to be non-None because a Region only exists
      for a Placement that has at least one Area. It is exposed as
      Optional for future flexibility (a Region could theoretically
      be derived from a non-Placement rule), but today it is always
      set.
    """

    region_id: str
    bounds: Bounds
    areas: tuple[Area, ...]
    semantic_role: Placement | None

    def __post_init__(self) -> None:
        if not self.region_id:
            raise ValueError("region_id must not be empty")
        if not self.areas:
            raise ValueError(f"Region {self.region_id!r} must contain at least one Area")

    def area_count(self) -> int:
        """Return the number of Areas in this Region."""
        return len(self.areas)

    def contains(self, area_id: str) -> bool:
        """Return True if the given area_id belongs to this Region."""
        return any(area.area_id == area_id for area in self.areas)

    def with_bounds(self, bounds: Bounds) -> Region:
        """Return a new Region with the given Bounds.

        Used by the LayoutRenderer / Workspace to produce a Region
        whose geometry is known, without mutating the original.
        """
        return Region(
            region_id=self.region_id,
            bounds=bounds,
            areas=self.areas,
            semantic_role=self.semantic_role,
        )

    def __repr__(self) -> str:
        return (
            f"Region(region_id={self.region_id!r}, "
            f"areas={[a.area_id for a in self.areas]!r}, "
            f"bounds={'placeholder' if self.bounds.is_placeholder() else self.bounds!r}, "
            f"semantic_role={self.semantic_role.value if self.semantic_role else None!r})"
        )

"""TopOverlay.

The TopOverlay is the first concrete Overlay in CranioZ. It
materializes the "Painel Superior" (Top Bar) shown across the UI
prototype: a horizontal bar anchored to the top of the Central
Region, with a fixed height and small lateral margins.

Geometric behavior (this piece):

- Anchored to the top of the target Region.
- Fixed height.
- Fixed lateral and top margins.
- Width spans the Region minus the lateral margins.

Content behavior (future pieces):

- The three segments (left, center, right) described in the
  prototype are not built here. This piece provides a container
  widget with an accessible content layout so that the Workspace
  (or a dedicated TopBar composition) can populate it later
  without changing the TopOverlay contract.

Design rules:

- The TopOverlay computes its own geometry from the Region Bounds
  it receives in ``reposition()``. It does not receive a
  pre-computed position.
- The overlay_id and target_region are class-level constants. Every
  TopOverlay instance targets the Central Region and has the same
  identity.
- The widget is a plain QFrame. Styling (colors, borders) will be
  applied later through the ThemeService, via Qt stylesheets keyed
  by the object name.
- The TopOverlay never destroys its widget. It can be hidden and
  shown again; the widget instance remains the same.
"""

from __future__ import annotations

from PySide6.QtWidgets import (
    QFrame,
    QHBoxLayout,
    QLayout,
    QWidget,
)

from cranioz.ui.workspace.areas.placement import Placement
from cranioz.ui.workspace.overlays.overlay import Overlay
from cranioz.ui.workspace.overlays.region import Bounds


class TopOverlayError(ValueError):
    """Raised when a TopOverlay is constructed with invalid parameters."""


class TopOverlay(Overlay):
    """Concrete Overlay implementing the Top Bar geometry.

    The TopOverlay is anchored to the top of the Central Region. Its
    height is fixed; its width follows the Region's width minus the
    configured lateral margins.
    """

    # ----- Class-level constants ------------------------------------

    OVERLAY_ID: str = "overlay.top_bar"
    TARGET_REGION: Placement = Placement.CENTRAL

    DEFAULT_HEIGHT: int = 40
    DEFAULT_MARGIN_X: int = 8
    DEFAULT_MARGIN_Y: int = 8

    # ----- Construction ---------------------------------------------

    def __init__(
        self,
        height: int = DEFAULT_HEIGHT,
        margin_x: int = DEFAULT_MARGIN_X,
        margin_y: int = DEFAULT_MARGIN_Y,
        parent: QWidget | None = None,
    ) -> None:
        if height <= 0:
            raise TopOverlayError(f"height must be positive, got {height}")
        if margin_x < 0:
            raise TopOverlayError(f"margin_x must be non-negative, got {margin_x}")
        if margin_y < 0:
            raise TopOverlayError(f"margin_y must be non-negative, got {margin_y}")

        self._height = height
        self._margin_x = margin_x
        self._margin_y = margin_y
        self._visible = True

        self._widget = QFrame(parent)
        self._widget.setObjectName("top_overlay")
        self._widget.setFixedHeight(height)

        # A horizontal layout is the natural container for the three
        # segments. The segments themselves are added by a future
        # piece; the layout is exposed via content_layout().
        self._layout = QHBoxLayout(self._widget)
        self._layout.setContentsMargins(4, 4, 4, 4)
        self._layout.setSpacing(4)

    # ----- Overlay API ----------------------------------------------

    @property
    def overlay_id(self) -> str:
        return self.OVERLAY_ID

    @property
    def target_region(self) -> Placement:
        return self.TARGET_REGION

    def widget(self) -> QWidget:
        return self._widget

    def reposition(self, bounds: Bounds) -> None:
        """Position the Top Bar over the given Region Bounds.

        The Top Bar is anchored to the top-left of the Region with
        the configured margins. Its width is the Region's width
        minus twice the lateral margin. If the resulting width
        would be negative (extremely narrow Region), it is clamped
        to zero so that Qt receives a valid geometry.
        """
        x = bounds.x + self._margin_x
        y = bounds.y + self._margin_y
        width = max(0, bounds.width - 2 * self._margin_x)
        height = self._height

        self._widget.setGeometry(
            round(x),
            round(y),
            round(width),
            round(height),
        )

    def set_visible(self, visible: bool) -> None:
        self._visible = visible
        self._widget.setVisible(visible)

    def is_visible(self) -> bool:
        return self._visible

    # ----- Convenience accessors ------------------------------------

    @property
    def height(self) -> int:
        """Return the fixed height of the Top Bar in pixels."""
        return self._height

    @property
    def margin_x(self) -> int:
        """Return the horizontal margin applied to the Top Bar."""
        return self._margin_x

    @property
    def margin_y(self) -> int:
        """Return the vertical margin applied to the Top Bar."""
        return self._margin_y

    def content_layout(self) -> QLayout:
        """Return the layout that will hold the Top Bar's content.

        Future pieces (the Workspace or a dedicated TopBar
        composition) add the left, center, and right segments to
        this layout. The TopOverlay itself does not create those
        segments: it only provides the container and its geometry.
        """
        return self._layout

"""EditorLayoutSpec.

Declarative description of how Editors are organized inside a single Area.

The same N-ary split model as the Area Layout Tree.

Rules:
- LEAF must have an editor_id and no children.
- SPLIT must have a direction and at least two children, and no editor_id.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from enum import Enum


class LayoutType(Enum):
    """Type of node in an Editor Layout Tree."""

    LEAF = "leaf"
    SPLIT = "split"


class SplitDirection(Enum):
    """Direction of a split node."""

    HORIZONTAL = "horizontal"
    VERTICAL = "vertical"


class EditorLayoutSpecError(ValueError):
    """Raised when an EditorLayoutSpec violates structural rules."""


@dataclass(frozen=True)
class EditorLayoutSpec:
    """Declarative description of an Editor Layout Tree node."""

    type: LayoutType
    direction: SplitDirection | None = None
    children: tuple[EditorLayoutSpec, ...] = field(default_factory=tuple)
    editor_id: str | None = None

    def __post_init__(self) -> None:
        self._validate()

    def _validate(self) -> None:
        if self.type is LayoutType.LEAF:
            if not self.editor_id:
                raise EditorLayoutSpecError("LEAF must declare editor_id")
            if self.children:
                raise EditorLayoutSpecError("LEAF must not have children")
            if self.direction is not None:
                raise EditorLayoutSpecError("LEAF must not declare direction")
        elif self.type is LayoutType.SPLIT:
            if self.editor_id is not None:
                raise EditorLayoutSpecError("SPLIT must not declare editor_id")
            if self.direction is None:
                raise EditorLayoutSpecError("SPLIT must declare direction")
            if len(self.children) < 2:
                raise EditorLayoutSpecError("SPLIT must have at least two children")
        else:
            raise EditorLayoutSpecError(f"unknown layout type: {self.type!r}")

    @classmethod
    def leaf(cls, editor_id: str) -> EditorLayoutSpec:
        """Create a LEAF node for the given editor_id."""
        return cls(type=LayoutType.LEAF, editor_id=editor_id)

    @classmethod
    def split(
        cls,
        direction: SplitDirection,
        children: tuple[EditorLayoutSpec, ...],
    ) -> EditorLayoutSpec:
        """Create a SPLIT node with the given direction and children."""
        return cls(type=LayoutType.SPLIT, direction=direction, children=children)

"""ModuleUISpec.

Declarative UI composition produced by a Module and materialized by the
Workspace. The Workspace does not know the Module that produced it.

Rules:
- At least one Area must have Placement. CENTRAL.
- Area ids must be unique within the spec.
- Overlays are declared separately.
"""

from __future__ import annotations

from dataclasses import dataclass, field

from cranioz.ui.workspace.areas.placement import Placement
from cranioz.ui.workspace.specs.area_spec import AreaSpec
from cranioz.ui.workspace.specs.overlay_spec import OverlaySpec


class ModuleUISpecError(ValueError):
    """Raised when a ModuleUISpec violates structural rules."""


@dataclass(frozen=True)
class ModuleUISpec:
    """Declarative UI composition produced by a Module."""

    areas: tuple[AreaSpec, ...]
    overlays: tuple[OverlaySpec, ...] = field(default_factory=tuple)

    def __post_init__(self) -> None:
        if not self.areas:
            raise ModuleUISpecError("ModuleUISpec must declare at least one Area")

        ids = [area.area_id for area in self.areas]
        if len(ids) != len(set(ids)):
            raise ModuleUISpecError("Area ids must be unique within a ModuleUISpec")

        if not any(area.placement is Placement.CENTRAL for area in self.areas):
            raise ModuleUISpecError("ModuleUISpec must declare at least one CENTRAL Area")
